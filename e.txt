local HitboxMap = {
    Head = "Head", Chest = "UpperTorso",
    Stomach = "LowerTorso", Body = "HumanoidRootPart"
}

local function GetPart(ch, name)
    local pn = HitboxMap[name] or "Head"
    return ch:FindFirstChild(pn) or ch:FindFirstChild("Head")
        or ch:FindFirstChild("UpperTorso") or ch:FindFirstChild("HumanoidRootPart")
end

local function CanSee(origin, part)
    if not S.WallCheck then return true end
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {LP.Character, Camera}
    local result = workspace:Raycast(origin, (part.Position - origin), params)
    if result then
        local model = part:FindFirstAncestorWhichIsA("Model")
        return model and result.Instance:IsDescendantOf(model)
    end
    return true
end

local function PredictPos(player, pos)
    if not S.Prediction then return pos end
    local vel = PlayerVelocities[player]
    if vel and vel.Magnitude > 1 then
        return pos + vel * (S.PredictionScale / 100)
    end
    return pos
end

local function GetBestTarget()
    local best, bestScore, bestPos = nil, math.huge, nil
    local center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    local camPos = Camera.CFrame.Position

    if S.StickyAim and StickyTarget and StickyTarget.Parent then
        local ch = StickyTarget.Character
        if ch then
            local hu = ch:FindFirstChildOfClass("Humanoid")
            local pt = GetPart(ch, S.Hitbox)
            if hu and hu.Health > 0 and pt then
                local d = (pt.Position - camPos).Magnitude
                if d <= S.MaxDistance then
                    local predicted = PredictPos(StickyTarget, pt.Position)
                    local sp, onScreen = Camera:WorldToViewportPoint(predicted)
                    if onScreen then
                        local sd = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                        if sd < S.FOV * 1.5 and CanSee(camPos, pt) then
                            return StickyTarget, predicted
                        end
                    end
                end
            end
        end
    end

    for _, player in ipairs(Players:GetPlayers()) do
        if player == LP then continue end
        if S.TeamCheck and player.Team == LP.Team then continue end
        local ch = player.Character
        if not ch then continue end
        local hu = ch:FindFirstChildOfClass("Humanoid")
        local pt = GetPart(ch, S.Hitbox)
        if not (hu and hu.Health > 0 and pt) then continue end
        local worldDist = (pt.Position - camPos).Magnitude
        if worldDist > S.MaxDistance then continue end
        if not CanSee(camPos, pt) then continue end

        local predicted = PredictPos(player, pt.Position)
        local sp, onScreen = Camera:WorldToViewportPoint(predicted)
        if onScreen then
            local screenDist = (Vector2.new(sp.X, sp.Y) - center).Magnitude
            if screenDist < S.FOV then
                local score = screenDist
                if score < bestScore then
                    bestScore = score
                    best = player
                    bestPos = predicted
                end
            end
        end
    end

    StickyTarget = best
    return best, bestPos
end

local ESPCache = {}

local function MakeLine(thick, col)
    local ok, l = pcall(Drawing.new, "Line")
    if ok and l then
        l.Thickness = thick or 1.5
        l.Color = col or Color3.new(1, 1, 1)
        l.Transparency = 1
        l.Visible = false
        return l
    end
end

local function MakeText(sz, col)
    local ok, t = pcall(Drawing.new, "Text")
    if ok and t then
        t.Size = sz or 13
        t.Color = col or Color3.new(1, 1, 1)
        t.Font = Drawing.Fonts.UI
        t.Outline = true
        t.OutlineColor = Color3.new(0, 0, 0)
        t.Center = true
        t.Visible = false
        return t
    end
end

local function MakeCircle(r, col, filled)
    local ok, c = pcall(Drawing.new, "Circle")
    if ok and c then
        c.Radius = r or 3
        c.Color = col or Color3.new(1, 1, 1)
        c.Filled = filled ~= false
        c.Thickness = 1.5
        c.Transparency = 1
        c.Visible = false
        return c
    end
end

local function MakeQuad()
    local ok, q = pcall(Drawing.new, "Quad")
    if ok and q then
        q.Visible = false
        q.Filled = true
        q.Transparency = 0.12
        return q
    end
end

local function BuildESP(player)
    if ESPCache[player] then return end
    local e = {}
    e.Box = {}; e.BoxOut = {}
    for i = 1, 4 do e.Box[i] = MakeLine(2); e.BoxOut[i] = MakeLine(4, Color3.new(0, 0, 0)) end
    e.Corner = {}; e.CornerOut = {}
    for i = 1, 8 do e.Corner[i] = MakeLine(2); e.CornerOut[i] = MakeLine(4, Color3.new(0, 0, 0)) end
    e.Box3D = {}
    for i = 1, 12 do e.Box3D[i] = MakeLine(1.2) end
    e.Fill = MakeQuad()
    e.Skel = {}
    for i = 1, 12 do e.Skel[i] = MakeLine(1.5) end
    e.NameTxt = MakeText(13)
    e.DistTxt = MakeText(11)
    e.WeaponTxt = MakeText(11)
    e.HpBGOut = MakeLine(8, Color3.fromRGB(10, 10, 10))
    e.HpBG = MakeLine(6, Color3.fromRGB(30, 30, 30))
    e.HpFill = MakeLine(4)
    e.HpTxt = MakeText(10)
    e.Tracer = MakeLine(1.5)
    e.TracerOut = MakeLine(3.5, Color3.new(0, 0, 0))
    e.HeadDot = MakeCircle(3, nil, true)
    e.HeadDotOut = MakeCircle(4.5, Color3.new(0, 0, 0), false)
    ESPCache[player] = e
end

local function DestroyESP(player)
    local e = ESPCache[player]
    if not e then return end
    local function rl(list) for _, d in ipairs(list) do if d then pcall(function() d:Remove() end) end end end
    rl(e.Box); rl(e.BoxOut); rl(e.Corner); rl(e.CornerOut); rl(e.Box3D); rl(e.Skel)
    for _, k in ipairs({"Fill","NameTxt","DistTxt","WeaponTxt","HpBGOut","HpBG","HpFill","HpTxt","Tracer","TracerOut","HeadDot","HeadDotOut"}) do
        if e[k] then pcall(function() e[k]:Remove() end) end
    end
    ESPCache[player] = nil
end

local function HideESP(e)
    if not e then return end
    local function hl(l) for _, d in ipairs(l) do if d then d.Visible = false end end end
    hl(e.Box); hl(e.BoxOut); hl(e.Corner); hl(e.CornerOut); hl(e.Box3D); hl(e.Skel)
    for _, k in ipairs({"Fill","NameTxt","DistTxt","WeaponTxt","HpBGOut","HpBG","HpFill","HpTxt","Tracer","TracerOut","HeadDot","HeadDotOut"}) do
        if e[k] then e[k].Visible = false end
    end
end

local CHDrawings = {}
local function ClearCH()
    for _, d in ipairs(CHDrawings) do pcall(function() d:Remove() end) end
    CHDrawings = {}
end

local function GenShape(t)
    if t == "Heart" then
        local pts = {}
        for a = 0, 2*math.pi, 0.15 do
            table.insert(pts, Vector2.new(14*math.sin(a)^3, -(11*math.cos(a)-4*math.cos(2*a)-2*math.cos(3*a)-math.cos(4*a))))
        end
        return pts, "dot"
    elseif t == "Star" then
        local pts = {}
        for i = 0, 9 do
            local a = math.rad(36*i-90); local r = i%2==0 and 16 or 7
            table.insert(pts, Vector2.new(math.cos(a)*r, math.sin(a)*r))
        end
        return pts, "line"
    elseif t == "Hexagon" then
        local pts = {}
        for i = 0, 5 do
            local a = math.rad(60*i-30)
            table.insert(pts, Vector2.new(math.cos(a)*14, math.sin(a)*14))
        end
        return pts, "line"
    elseif t == "Square" then
        local s = 10
        return {Vector2.new(-s,-s),Vector2.new(s,-s),Vector2.new(s,s),Vector2.new(-s,s)}, "line"
    elseif t == "Dot" then
        return {Vector2.new(0,0)}, "dot"
    end
    return {Vector2.new(0,0)}, "dot"
end

local chPts, chMode, chDots, chLines = {}, "dot", {}, {}
local function RebuildCH()
    ClearCH(); chDots = {}; chLines = {}
    chPts, chMode = GenShape(S.CrosshairType)
    if chMode == "dot" then
        for i = 1, #chPts do
            local ok, d = pcall(Drawing.new, "Circle")
            if ok and d then d.Filled = true; d.Visible = false; table.insert(CHDrawings, d); table.insert(chDots, d) end
        end
    else
        for i = 1, #chPts do
            local ok, l = pcall(Drawing.new, "Line")
            if ok and l then l.Visible = false; table.insert(CHDrawings, l); table.insert(chLines, l) end
        end
    end
end
RebuildCH()

-- ═══════════ CHAMS ═══════════
local Chams, Glows = {}, {}
local function ApplyChams(p)
    if not p.Character then return end
    if Chams[p] then pcall(function() Chams[p]:Destroy() end) end
    pcall(function()
        local h = Instance.new("Highlight")
        h.FillColor = Theme.Accent; h.OutlineColor = Color3.new(1,1,1)
        h.FillTransparency = 0.5; h.OutlineTransparency = 0.3
        h.DepthMode = S.ChamsMode == "Wall" and Enum.HighlightDepthMode.AlwaysOnTop or Enum.HighlightDepthMode.Occluded
        h.Adornee = p.Character; h.Parent = p.Character; Chams[p] = h
    end)
end
local function RemoveChams(p) if Chams[p] then pcall(function() Chams[p]:Destroy() end); Chams[p] = nil end end
local function RefreshChams()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP then RemoveChams(p); if S.ChamsEnabled then ApplyChams(p) end end
    end
end

local function ApplyWeather()
    SafeSet("ClockTime", S.TimeOfDay)
    if S.FullBright then
        SafeSet("Ambient", Color3.new(1,1,1)); SafeSet("OutdoorAmbient", Color3.new(1,1,1))
        SafeSet("Brightness", 3); SafeSet("GlobalShadows", false)
    elseif S.NightMode then
        SafeSet("Ambient", Color3.fromRGB(0,0,30)); SafeSet("OutdoorAmbient", Color3.fromRGB(0,0,30))
        SafeSet("Brightness", 0.2)
    else
        SafeSet("Ambient", Color3.fromRGB(S.AmbientR, S.AmbientG, S.AmbientB))
        SafeSet("OutdoorAmbient", Color3.fromRGB(S.AmbientR, S.AmbientG, S.AmbientB))
        SafeSet("Brightness", S.Brightness)
    end
    SafeSet("ColorShift_Top", Color3.fromRGB(S.SkyR, S.SkyG, S.SkyB))
    SafeSet("ColorShift_Bottom", Color3.fromRGB(S.SkyR, S.SkyG, S.SkyB))
    if S.FogEnabled then
        SafeSet("FogColor", Color3.fromRGB(S.FogR, S.FogG, S.FogB))
        SafeSet("FogStart", S.FogStart); SafeSet("FogEnd", S.FogEnd)
    else SafeSet("FogEnd", 100000); SafeSet("FogStart", 0) end
    Camera.FieldOfView = S.CameraFOV
    pcall(function()
        if S.ThirdPerson then
            LP.CameraMinZoomDistance = S.ThirdPersonDist; LP.CameraMaxZoomDistance = S.ThirdPersonDist
        else LP.CameraMinZoomDistance = 0.5; LP.CameraMaxZoomDistance = 128 end
    end)
end

    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.ZIndex = 22
    lbl.Parent = card

    -- Toggle Switch (Pill shape with sliding dot)
    local toggleFrame = Instance.new("TextButton")
    toggleFrame.Size = UDim2.new(0, 46, 0, 24)
    toggleFrame.Position = UDim2.new(1, -58, 0.5, -12)
    toggleFrame.BackgroundColor3 = default and Theme.Accent or Theme.ToggleOff
    toggleFrame.Text = ""
    toggleFrame.BorderSizePixel = 0
    toggleFrame.ZIndex = 23
    toggleFrame.Parent = card
    Instance.new("UICorner", toggleFrame).CornerRadius = UDim.new(1, 0)

    local dot = Instance.new("Frame")
    dot.Size = UDim2.new(0, 18, 0, 18)
    dot.Position = default and UDim2.new(1, -21, 0.5, -9) or UDim2.new(0, 3, 0.5, -9)
    dot.BackgroundColor3 = Theme.ToggleDot
    dot.BorderSizePixel = 0
    dot.ZIndex = 24
    dot.Parent = toggleFrame
    Instance.new("UICorner", dot).CornerRadius = UDim.new(1, 0)

    -- Drop shadow on dot
    local shadow = Instance.new("UIStroke", dot)
    shadow.Color = Color3.new(0, 0, 0); shadow.Thickness = 1; shadow.Transparency = 0.7

    local state = default
    toggleFrame.MouseButton1Click:Connect(function()
        state = not state
        TweenService:Create(toggleFrame, TweenInfo.new(0.2, Enum.EasingStyle.Quad), {
            BackgroundColor3 = state and Theme.Accent or Theme.ToggleOff
        }):Play()
        TweenService:Create(dot, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.InOut), {
            Position = state and UDim2.new(1, -21, 0.5, -9) or UDim2.new(0, 3, 0.5, -9)
        }):Play()
        callback(state)
    end)

    return toggleFrame
end

local function MakeSlider(parent, text, y, mn, mx, default, callback)
    local card = MakeCard(parent, y, 58)

    local lbl = Instance.new("TextLabel")
    lbl.Size = UDim2.new(0.6, 0, 0, 22)
    lbl.Position = UDim2.new(0, 16, 0, 4)
    lbl.BackgroundTransparency = 1
    lbl.Text = text
    lbl.TextColor3 = Theme.Text
    lbl.TextSize = IS_MOBILE and 11 or 12
    lbl.Font = Enum.Font.GothamSemibold
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.ZIndex = 22
    lbl.Parent = card

    local valLbl = Instance.new("TextLabel")
    valLbl.Size = UDim2.new(0.3, 0, 0, 22)
    valLbl.Position = UDim2.new(0.7, -10, 0, 4)
    valLbl.BackgroundTransparency = 1
    valLbl.Text = tostring(default)
    valLbl.TextColor3 = Theme.Accent
    valLbl.TextSize = 12
    valLbl.Font = Enum.Font.GothamBold
    valLbl.TextXAlignment = Enum.TextXAlignment.Right
    valLbl.ZIndex = 22
    valLbl.Parent = card

    local track = Instance.new("Frame")
    track.Size = UDim2.new(1, -32, 0, 6)
    track.Position = UDim2.new(0, 16, 0, 34)
    track.BackgroundColor3 = Theme.ToggleOff
    track.BorderSizePixel = 0
    track.ZIndex = 22
    track.Parent = card
    Instance.new("UICorner", track).CornerRadius = UDim.new(1, 0)

    local fill = Instance.new("Frame")
    fill.Size = UDim2.new((default-mn)/(mx-mn), 0, 1, 0)
    fill.BackgroundColor3 = Theme.Accent
    fill.BorderSizePixel = 0
    fill.ZIndex = 23
    fill.Parent = track
    Instance.new("UICorner", fill).CornerRadius = UDim.new(1, 0)

    local knob = Instance.new("Frame")
    knob.Size = UDim2.new(0, 14, 0, 14)
    knob.Position = UDim2.new((default-mn)/(mx-mn), -7, 0.5, -7)
    knob.BackgroundColor3 = Color3.new(1,1,1)
    knob.BorderSizePixel = 0
    knob.ZIndex = 24
    knob.Parent = track
    Instance.new("UICorner", knob).CornerRadius = UDim.new(1, 0)
    Instance.new("UIStroke", knob).Color = Theme.Accent

    local dragging = false
    local function update(x)
        local r = math.clamp((x - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
        local val = math.floor(mn + (mx-mn)*r)
        fill.Size = UDim2.new(r, 0, 1, 0)
        knob.Position = UDim2.new(r, -7, 0.5, -7)
        valLbl.Text = tostring(val)
        callback(val)
    end
    track.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = true; update(i.Position.X) end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then update(i.Position.X) end
    end)
    UIS.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = false end
    end)
end

local function MakeDropdown(parent, text, y, opts, default, callback)
    local card = MakeCard(parent, y, 48)

    Instance.new("TextLabel", card).Size = UDim2.new(0.5, 0, 1, 0)
    local lbl = card:FindFirstChildOfClass("TextLabel")
    lbl.Position = UDim2.new(0, 16, 0, 0)
    lbl.BackgroundTransparency = 1; lbl.Text = text
    lbl.TextColor3 = Theme.Text; lbl.TextSize = IS_MOBILE and 11 or 12
    lbl.Font = Enum.Font.GothamSemibold; lbl.TextXAlignment = Enum.TextXAlignment.Left; lbl.ZIndex = 22

    local idx = 1
    for i, o in ipairs(opts) do if o == default then idx = i end end

    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0, IS_MOBILE and 110 or 130, 0, 28)
    btn.Position = UDim2.new(1, -(IS_MOBILE and 118 or 138), 0.5, -14)
    btn.BackgroundColor3 = Theme.ToggleOff
    btn.Text = "  " .. opts[idx] .. "  ▼"
    btn.TextColor3 = Theme.Accent
    btn.TextSize = 11; btn.Font = Enum.Font.GothamBold
    btn.BorderSizePixel = 0; btn.ZIndex = 23; btn.Parent = card
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 6)

    btn.MouseButton1Click:Connect(function()
        idx = idx % #opts + 1
        btn.Text = "  " .. opts[idx] .. "  ▼"
        callback(opts[idx])
    end)
end

    end)
end

local function SetupDrag(handle, frame)
    local dr, ds, sp = false, nil, nil
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dr = true; ds = i.Position; sp = frame.Position
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dr and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            local d = i.Position - ds
            frame.Position = UDim2.new(sp.X.Scale, sp.X.Offset+d.X, sp.Y.Scale, sp.Y.Offset+d.Y)
        end
    end)
    UIS.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dr = false end
    end)
end
SetupDrag(TitleBar, MainFrame)

-- ═══════════ MAIN RENDER LOOP ═══════════
local chRot = 0
local velTimer = 0
local tt = 0

RunService.RenderStepped:Connect(function(dt)
    tt = tt + dt
    local center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    local camPos = Camera.CFrame.Position

    -- FPS
    fpsCount = fpsCount + 1; fpsTime = fpsTime + dt
    if fpsTime >= 0.5 then currentFPS = math.floor(fpsCount/fpsTime); fpsCount = 0; fpsTime = 0 end

    -- Velocity tracking
    velTimer = velTimer + dt
    if velTimer >= 0.033 then
        velTimer = 0
        for _, p in ipairs(Players:GetPlayers()) do
            if p == LP then continue end
            local ch = p.Character
            if ch then
                local hrp = ch:FindFirstChild("HumanoidRootPart")
                if hrp then
                    local last = PlayerLastPos[p]
                    PlayerVelocities[p] = last and (hrp.Position - last) / 0.033 or Vector3.zero
                    PlayerLastPos[p] = hrp.Position
                end
            end
        end
    end

    -- Movement
    local myChar = LP.Character
    if myChar then
        local hu = myChar:FindFirstChildOfClass("Humanoid")
        local hrp = myChar:FindFirstChild("HumanoidRootPart")
        if hu and hrp then
            -- Bunny Hop
            if S.BunnyHop then
                if hu:GetState() == Enum.HumanoidStateType.Landed or hu.FloorMaterial ~= Enum.Material.Air then
                    hu.Jump = true
                end
            end
            -- Speed/Jump reapply
            if S.SpeedBoost then hu.WalkSpeed = S.SpeedValue end
            if S.JumpBoost then hu.JumpPower = S.JumpValue; hu.UseJumpPower = true end
        end
    end

    -- Toggle button pulse
    tbStroke.Thickness = 2 + math.sin(tt * 3) * 0.8

    -- FOV Circle
    if FOVInner then
        FOVInner.Position = center; FOVInner.Radius = S.FOV; FOVInner.Color = C3(S.FOVColor); FOVInner.Visible = S.ShowFOV and S.AimbotEnabled
    end
    if FOVOuter then
        FOVOuter.Position = center; FOVOuter.Radius = S.FOV; FOVOuter.Visible = S.ShowFOV and S.AimbotEnabled
    end

    -- ═══ CROSSHAIR ═══
    if S.CrosshairRotate then chRot = (chRot + dt * S.CrosshairSpeed) % 360 end
    local rot = math.rad(chRot)
    local szM = S.CrosshairSize / 100
    local thM = S.CrosshairThickness / 100
    local opa = S.CrosshairOpacity / 100
    local cCol = C3(S.CrosshairColor)
    local vis = S.ShowCrosshair and opa > 0.01

    if chMode == "dot" then
        for i, d in ipairs(chDots) do
            if chPts[i] then
                local pt = chPts[i]
                d.Position = center + Rot2(Vector2.new(pt.X * szM, pt.Y * szM), rot)
                d.Radius = S.CrosshairType == "Dot" and S.CrosshairSize / 25 or 1.5 * thM
                d.Color = cCol; d.Transparency = opa; d.Visible = vis
            else d.Visible = false end
        end
        for _, l in ipairs(chLines) do l.Visible = false end
    else
        for _, d in ipairs(chDots) do d.Visible = false end
        for i, l in ipairs(chLines) do
            if chPts[i] then
                local ni = i % #chPts + 1
                local p1, p2 = chPts[i], chPts[ni]
                l.From = center + Rot2(Vector2.new(p1.X * szM, p1.Y * szM), rot)
                l.To = center + Rot2(Vector2.new(p2.X * szM, p2.Y * szM), rot)
                l.Color = cCol; l.Thickness = 2 * thM; l.Transparency = opa; l.Visible = vis
            else l.Visible = false end
        end
    end

    -- ═══ 0ms AIMBOT ═══
    local shouldAim = false
    if S.AimbotEnabled then
        if S.AimMode == "Always" then shouldAim = true
        elseif S.AimMode == "Hold" then shouldAim = AimbotActive
        elseif S.AimMode == "Toggle" then shouldAim = ToggleAim end
    end

    if shouldAim then
        local target, aimPos = GetBestTarget()
        if target and aimPos then
            local targetCF = CFrame.new(camPos, aimPos)

            if S.Smoothness <= 0 or S.SnapAim then
                -- 0ms instant lock
                Camera.CFrame = targetCF
            else
                local alpha = math.clamp(dt * (22 - S.Smoothness) * 5, 0, 1)
                Camera.CFrame = Camera.CFrame:Lerp(targetCF, alpha)
            end
        end
    end

    -- ═══ RADAR ═══
    if S.RadarEnabled and RadarF.Visible then
        local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
        for p, dot in pairs(RadarDots) do
            if not p.Parent or not p.Character then dot.Visible = false
            else
                local root = p.Character:FindFirstChild("HumanoidRootPart")
                if root and myHrp then
                    local rel = root.Position - myHrp.Position
                    local look = Vector3.new(Camera.CFrame.LookVector.X, 0, Camera.CFrame.LookVector.Z)
                    if look.Magnitude > 0 then look = look.Unit end
                    local ang = math.atan2(look.X, look.Z)
                    local cs, sn = math.cos(-ang), math.sin(-ang)
                    local rx, rz = rel.X*cs - rel.Z*sn, rel.X*sn + rel.Z*cs
                    dot.Position = UDim2.new(math.clamp(0.5+rx/S.RadarZoom/2,0,1),-2, math.clamp(0.5+rz/S.RadarZoom/2,0,1),-2)
                    dot.BackgroundColor3 = ESPColor(p); dot.Visible = true
                else dot.Visible = false end
            end
        end
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LP and not RadarDots[p] then
                local d = Instance.new("Frame"); d.Size = UDim2.new(0,5,0,5)
                d.BackgroundColor3 = C3(S.ESPEnemyColor); d.BorderSizePixel = 0; d.ZIndex = 102; d.Parent = RadarF
                Instance.new("UICorner", d).CornerRadius = UDim.new(1, 0); RadarDots[p] = d
            end
        end
    else for _, d in pairs(RadarDots) do d.Visible = false end end

    -- ═══ ULTRA ESP RENDERING ═══
    for player, e in pairs(ESPCache) do
        if not player or not player.Parent then HideESP(e); continue end
        if not S.ESPEnabled then HideESP(e); continue end
        local ch = player.Character
        if not ch then HideESP(e); continue end
        local hu = ch:FindFirstChildOfClass("Humanoid")
        local root = ch:FindFirstChild("HumanoidRootPart")
        local head = ch:FindFirstChild("Head")
        if not (hu and hu.Health > 0 and root and head) then HideESP(e); continue end

        local wDist = (root.Position - camPos).Magnitude
        if wDist > S.ESPMaxDist then HideESP(e); continue end

        local htop = head.Position + Vector3.new(0, head.Size.Y/2 + 0.3, 0)
        local hbot = root.Position - Vector3.new(0, 3, 0)
        local hs, hv = Camera:WorldToViewportPoint(htop)
        local rs, rv = Camera:WorldToViewportPoint(hbot)
        if not (hv and rv) then HideESP(e); continue end

        local top, bot = hs.Y, rs.Y
        local h = math.max(bot - top, 8)
        local cx = (hs.X + rs.X) / 2
        local w = math.max(h * 0.5, 12)
        local col = ESPColor(player)
        local thick = S.ESPThickness

        -- Reset visibility
        for _, l in ipairs(e.Box) do l.Visible = false end
        for _, l in ipairs(e.BoxOut) do l.Visible = false end
        for _, l in ipairs(e.Corner) do l.Visible = false end
        for _, l in ipairs(e.CornerOut) do l.Visible = false end
        for _, l in ipairs(e.Box3D) do l.Visible = false end
        if e.Fill then e.Fill.Visible = false end

        if S.ShowBoxes then
            if S.BoxStyle == "2D" then
                local tl = Vector2.new(cx-w/2, top)
                local tr = Vector2.new(cx+w/2, top)
                local bl = Vector2.new(cx-w/2, bot)
                local br = Vector2.new(cx+w/2, bot)
                local pts = {tl, tr, br, bl}
                for i = 1, 4 do
                    local ni = i%4+1
                    if S.ESPOutline and e.BoxOut[i] then
                        e.BoxOut[i].From = pts[i]; e.BoxOut[i].To = pts[ni]
                        e.BoxOut[i].Thickness = thick + 2; e.BoxOut[i].Visible = true
                    end
                    e.Box[i].From = pts[i]; e.Box[i].To = pts[ni]
                    e.Box[i].Color = col; e.Box[i].Thickness = thick; e.Box[i].Visible = true
                end
                if S.ESPFill and e.Fill then
                    e.Fill.PointA = tl; e.Fill.PointB = tr; e.Fill.PointC = br; e.Fill.PointD = bl
                    e.Fill.Color = col; e.Fill.Transparency = 0.12; e.Fill.Visible = true
                end
            elseif S.BoxStyle == "Corner" then
                local cl = math.min(w, h) * 0.28
                local corners = {
                    {Vector2.new(cx-w/2,top), Vector2.new(cx-w/2+cl,top)},
                    {Vector2.new(cx-w/2,top), Vector2.new(cx-w/2,top+cl)},
                    {Vector2.new(cx+w/2,top), Vector2.new(cx+w/2-cl,top)},
                    {Vector2.new(cx+w/2,top), Vector2.new(cx+w/2,top+cl)},
                    {Vector2.new(cx-w/2,bot), Vector2.new(cx-w/2+cl,bot)},
                    {Vector2.new(cx-w/2,bot), Vector2.new(cx-w/2,bot-cl)},
                    {Vector2.new(cx+w/2,bot), Vector2.new(cx+w/2-cl,bot)},
                    {Vector2.new(cx+w/2,bot), Vector2.new(cx+w/2,bot-cl)},
                }
                for i = 1, 8 do
                    if S.ESPOutline and e.CornerOut[i] then
                        e.CornerOut[i].From = corners[i][1]; e.CornerOut[i].To = corners[i][2]
                        e.CornerOut[i].Thickness = thick + 2; e.CornerOut[i].Visible = true
                    end
                    e.Corner[i].From = corners[i][1]; e.Corner[i].To = corners[i][2]
                    e.Corner[i].Color = col; e.Corner[i].Thickness = thick; e.Corner[i].Visible = true
                end
            elseif S.BoxStyle == "3D" then
                local ext = Vector3.new(2, 3, 1.5)
                local c3d = {
                    root.Position+Vector3.new(-ext.X,ext.Y,-ext.Z), root.Position+Vector3.new(ext.X,ext.Y,-ext.Z),
                    root.Position+Vector3.new(ext.X,ext.Y,ext.Z), root.Position+Vector3.new(-ext.X,ext.Y,ext.Z),
                    root.Position+Vector3.new(-ext.X,-ext.Y,-ext.Z), root.Position+Vector3.new(ext.X,-ext.Y,-ext.Z),
                    root.Position+Vector3.new(ext.X,-ext.Y,ext.Z), root.Position+Vector3.new(-ext.X,-ext.Y,ext.Z),
                }
                local sp = {}; local allOn = true
                for i, c in ipairs(c3d) do
                    local s, o = Camera:WorldToViewportPoint(c); sp[i] = Vector2.new(s.X, s.Y); if not o then allOn = false end
                end
                if allOn then
                    local edges = {{1,2},{2,3},{3,4},{4,1},{5,6},{6,7},{7,8},{8,5},{1,5},{2,6},{3,7},{4,8}}
                    for i, ed in ipairs(edges) do
                        e.Box3D[i].From = sp[ed[1]]; e.Box3D[i].To = sp[ed[2]]
                        e.Box3D[i].Color = col; e.Box3D[i].Thickness = 1.2; e.Box3D[i].Visible = true
                    end
                end
            end
        end

        -- Skeleton
        for _, l in ipairs(e.Skel) do l.Visible = false end
        if S.ShowSkeleton then
            for i, bone in ipairs(R15Bones) do
                if i > #e.Skel then break end
                local p1 = ch:FindFirstChild(bone[1]); local p2 = ch:FindFirstChild(bone[2])
                if p1 and p2 then
                    local s1, o1 = Camera:WorldToViewportPoint(p1.Position)
                    local s2, o2 = Camera:WorldToViewportPoint(p2.Position)
                    if o1 and o2 then
                        e.Skel[i].From = Vector2.new(s1.X, s1.Y); e.Skel[i].To = Vector2.new(s2.X, s2.Y)
                        e.Skel[i].Color = C3(S.SkeletonColor); e.Skel[i].Visible = true
                    end
                end
            end
        end

        -- Name
        if S.ShowNames and e.NameTxt then
            e.NameTxt.Text = player.DisplayName
            e.NameTxt.Position = Vector2.new(cx, top - 16)
            e.NameTxt.Color = col; e.NameTxt.Size = S.ESPTextSize; e.NameTxt.Visible = true
        elseif e.NameTxt then e.NameTxt.Visible = false end

        -- Health
        if e.HpBGOut then e.HpBGOut.Visible = false end
        if e.HpBG then e.HpBG.Visible = false end
        if e.HpFill then e.HpFill.Visible = false end
        if e.HpTxt then e.HpTxt.Visible = false end
        if S.ShowHealth then
            local hp = math.clamp(hu.Health / hu.MaxHealth, 0, 1)
            local hpCol = Color3.fromRGB(math.floor((1-hp)*255), math.floor(hp*255), 30)
            local bx = cx - w/2 - 7
            e.HpBGOut.From = Vector2.new(bx, top-1); e.HpBGOut.To = Vector2.new(bx, bot+1); e.HpBGOut.Visible = true
            e.HpBG.From = Vector2.new(bx, top); e.HpBG.To = Vector2.new(bx, bot); e.HpBG.Visible = true
            e.HpFill.Color = hpCol
            e.HpFill.From = Vector2.new(bx, bot); e.HpFill.To = Vector2.new(bx, bot - h*hp); e.HpFill.Visible = true
        end

        -- Distance
        if S.ShowDistance and e.DistTxt then
            e.DistTxt.Text = "["..math.floor(wDist).."m]"
            e.DistTxt.Position = Vector2.new(cx, bot + 4)
            e.DistTxt.Color = col; e.DistTxt.Visible = true
        elseif e.DistTxt then e.DistTxt.Visible = false end

        -- Weapon
        if S.ShowWeapon and e.WeaponTxt then
            local tool = ch:FindFirstChildOfClass("Tool")
            if tool then
                e.WeaponTxt.Text = "["..tool.Name.."]"
                e.WeaponTxt.Position = Vector2.new(cx, bot + (S.ShowDistance and 18 or 4))
                e.WeaponTxt.Visible = true
            else e.WeaponTxt.Visible = false end
        elseif e.WeaponTxt then e.WeaponTxt.Visible = false end

        -- Tracers
        if S.ShowTracers then
            if S.ESPOutline and e.TracerOut then
                e.TracerOut.From = Vector2.new(center.X, Camera.ViewportSize.Y)
                e.TracerOut.To = Vector2.new(cx, bot); e.TracerOut.Visible = true
            end
            if e.Tracer then
                e.Tracer.From = Vector2.new(center.X, Camera.ViewportSize.Y)
                e.Tracer.To = Vector2.new(cx, bot); e.Tracer.Color = col; e.Tracer.Visible = true
            end
        else
            if e.Tracer then e.Tracer.Visible = false end
            if e.TracerOut then e.TracerOut.Visible = false end
        end

        -- Head Dot
        if S.ShowHeadDot then
            local headSP, headOn = Camera:WorldToViewportPoint(head.Position)
            if headOn then
                local hp2 = Vector2.new(headSP.X, headSP.Y)
                if e.HeadDotOut and S.ESPOutline then
                    e.HeadDotOut.Position = hp2; e.HeadDotOut.Radius = 4.5; e.HeadDotOut.Visible = true; e.HeadDotOut.Filled = false
                end
                if e.HeadDot then
                    e.HeadDot.Position = hp2; e.HeadDot.Radius = 3; e.HeadDot.Color = col; e.HeadDot.Visible = true
                end
            else
                if e.HeadDot then e.HeadDot.Visible = false end
                if e.HeadDotOut then e.HeadDotOut.Visible = false end
            end
        else
            if e.HeadDot then e.HeadDot.Visible = false end
            if e.HeadDotOut then e.HeadDotOut.Visible = false end
        end
    end
end)
