local function WorldToScreen(position)
    local screenPos, onScreen = Camera:WorldToViewportPoint(position)
    return Vector2.new(screenPos.X, screenPos.Y), onScreen, screenPos.Z
end

local function GetDistance(pos1, pos2)
    return (pos1 - pos2).Magnitude
end

local function GetCharacter(player)
    return player and player.Character
end

local function GetHumanoid(character)
    return character and character:FindFirstChildOfClass("Humanoid")
end

local function GetRootPart(character)
    return character and (character:FindFirstChild("HumanoidRootPart") or character.PrimaryPart)
end

local function GetHead(character)
    return character and character:FindFirstChild("Head")
end

local function IsAlive(player)
    local char = GetCharacter(player)
    local humanoid = GetHumanoid(char)
    local root = GetRootPart(char)
    return char and humanoid and root and humanoid.Health > 0
end

local function IsSameTeam(player)
    if not Config.ESP.TeamCheck then return false end
    if LocalPlayer.Team and player.Team then
        return LocalPlayer.Team == player.Team
    end
    return false
end

local function GetBodyPart(character, partName)
    if partName == "Голова" then
        return character:FindFirstChild("Head")
    elseif partName == "Корпус" then
        return character:FindFirstChild("UpperTorso") or character:FindFirstChild("Torso")
    elseif partName == "Ноги" then
        return character:FindFirstChild("LeftFoot") or character:FindFirstChild("LeftLeg") or character:FindFirstChild("Left Leg")
    elseif partName == "Всё тело" then
        return character:FindFirstChild("HumanoidRootPart")
    end
    return character:FindFirstChild("Head")
end

local function Lerp(a, b, t)
    return a + (b - a) * t
end

local function CleanDrawing(obj)
    if obj then
        pcall(function() obj:Remove() end)
    end
end

local function SwitchTab(tabName)
    CurrentTab = tabName
    for name, btn in pairs(TabButtons) do
        if name == tabName then
            btn.BackgroundColor3 = Colors.TabActive
            btn.TextColor3 = Colors.TextHighlight
        else
            btn.BackgroundColor3 = Colors.TabInactive
            btn.TextColor3 = Colors.TextSecondary
        end
    end
    for name, page in pairs(TabPages) do
        page.Visible = (name == tabName)
    end
    ContentFrame.CanvasPosition = Vector2.new(0, 0)
end

        if callback then callback(currentValue) end
    end

    sliderInput.MouseButton1Down:Connect(function()
        dragging = true
    end)

    table.insert(InputConnections, UserInputService.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            UpdateSlider(input)
        end
    end))

    table.insert(InputConnections, UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end))

    sliderInput.MouseButton1Down:Connect(function()
        dragging = true
        local input = {Position = Vector3.new(Mouse.X, Mouse.Y, 0)}
        UpdateSlider(input)
    end)

    return {
        SetValue = function(val)
            currentValue = math.clamp(val, min, max)
            local percent = (currentValue - min) / (max - min)
            sliderFill.Size = UDim2.new(percent, 0, 1, 0)
            sliderKnob.Position = UDim2.new(percent, -8, 0.5, -8)
            valueLabel.Text = tostring(math.floor(currentValue))
        end,
        GetValue = function() return currentValue end,
        SetMax = function(newMax)
            max = newMax
            local percent = math.clamp((currentValue - min) / (max - min), 0, 1)
            sliderFill.Size = UDim2.new(percent, 0, 1, 0)
            sliderKnob.Position = UDim2.new(percent, -8, 0.5, -8)
        end
    }
end

    for i, opt in ipairs(options) do
        local btn = Instance.new("TextButton")
        btn.Name = "Opt_" .. opt
        btn.Size = UDim2.new(0, btnWidth, 1, 0)
        btn.BackgroundColor3 = (i == selectedIndex) and Colors.AccentPrimary or Colors.AccentDark
        btn.BorderSizePixel = 0
        btn.Text = opt
        btn.TextColor3 = (i == selectedIndex) and Colors.TextHighlight or Colors.TextMuted
        btn.TextSize = 10
        btn.Font = Enum.Font.GothamSemibold
        btn.AutoButtonColor = false
        btn.ZIndex = 54
        btn.TextTruncate = Enum.TextTruncate.AtEnd
        btn.Parent = btnContainer

        local c = Instance.new("UICorner")
        c.CornerRadius = UDim.new(0, 4)
        c.Parent = btn

        optionBtns[i] = btn

        btn.MouseButton1Click:Connect(function()
            selectedIndex = i
            for j, b in ipairs(optionBtns) do
                b.BackgroundColor3 = (j == i) and Colors.AccentPrimary or Colors.AccentDark
                b.TextColor3 = (j == i) and Colors.TextHighlight or Colors.TextMuted
            end
            if callback then callback(opt) end
        end)
    end

    return {
        GetValue = function() return options[selectedIndex] end,
        SetValue = function(val)
            for i, v in ipairs(options) do
                if v == val then
                    selectedIndex = i
                    for j, b in ipairs(optionBtns) do
                        b.BackgroundColor3 = (j == i) and Colors.AccentPrimary or Colors.AccentDark
                        b.TextColor3 = (j == i) and Colors.TextHighlight or Colors.TextMuted
                    end
                    break
                end
            end
        end
    }
end

local function RemoveESPForPlayer(player)
    local espData = ESPObjects[player]
    if not espData then return end

    CleanDrawing(espData.Box)
    CleanDrawing(espData.BoxOutline)
    CleanDrawing(espData.NameText)
    CleanDrawing(espData.DistText)
    CleanDrawing(espData.HeadDot)
    CleanDrawing(espData.Tracer)
    CleanDrawing(espData.HealthBarBG)
    CleanDrawing(espData.HealthBarFill)

    for _, line in pairs(espData.SkeletonLines) do
        CleanDrawing(line)
    end

    ESPObjects[player] = nil
end

local function UpdateESP()
    if not Config.ESP.Enabled then
        for _, espData in pairs(ESPObjects) do
            HideESP(espData)
        end
        return
    end

    local localChar = GetCharacter(LocalPlayer)
    local localRoot = localChar and GetRootPart(localChar)
    if not localRoot then return end

    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer then continue end

        if not ESPObjects[player] then
            CreateESPForPlayer(player)
        end

        local espData = ESPObjects[player]
        local character = GetCharacter(player)
        local humanoid = character and GetHumanoid(character)
        local rootPart = character and GetRootPart(character)
        local head = character and GetHead(character)

        if not (character and humanoid and rootPart and head and humanoid.Health > 0) then
            HideESP(espData)
            continue
        end

        local distance = GetDistance(localRoot.Position, rootPart.Position)

        if distance > Config.ESP.MaxDistance then
            HideESP(espData)
            continue
        end

        local isFriendly = IsSameTeam(player)
        local espColor = isFriendly and Colors.TeamFriendly or Colors.TeamEnemy

        -- Основные экранные координаты
        local rootScreen, rootOnScreen, rootDepth = WorldToScreen(rootPart.Position)
        local headScreen, headOnScreen = WorldToScreen(head.Position + Vector3.new(0, 0.5, 0))
        local feetScreen = WorldToScreen(rootPart.Position - Vector3.new(0, 3, 0))

        if not rootOnScreen then
            HideESP(espData)
            continue
        end

        local boxHeight = math.abs(headScreen.Y - feetScreen.Y)
        local boxWidth = boxHeight * 0.55
        local boxTopLeft = Vector2.new(rootScreen.X - boxWidth / 2, headScreen.Y)

        -- Box ESP
        if Config.ESP.Box then
            espData.BoxOutline.Size = Vector2.new(boxWidth, boxHeight)
            espData.BoxOutline.Position = boxTopLeft
            espData.BoxOutline.Visible = true

            espData.Box.Size = Vector2.new(boxWidth, boxHeight)
            espData.Box.Position = boxTopLeft
            espData.Box.Color = espColor
            espData.Box.Visible = true
        else
            espData.Box.Visible = false
            espData.BoxOutline.Visible = false
        end

        -- Name ESP
        if Config.ESP.Name then
            espData.NameText.Position = Vector2.new(rootScreen.X, headScreen.Y - 18)
            espData.NameText.Text = player.DisplayName or player.Name
            espData.NameText.Color = espColor
            espData.NameText.Visible = true
        else
            espData.NameText.Visible = false
        end

        -- Distance ESP
        if Config.ESP.Distance then
            espData.DistText.Position = Vector2.new(rootScreen.X, feetScreen.Y + 4)
            espData.DistText.Text = math.floor(distance) .. " м"
            espData.DistText.Color = Colors.TextSecondary
            espData.DistText.Visible = true
        else
            espData.DistText.Visible = false
        end

        -- Health Bar ESP
        if Config.ESP.HealthBar then
            local healthPercent = math.clamp(humanoid.Health / humanoid.MaxHealth, 0, 1)
            local barX = boxTopLeft.X - 6
            local barY = headScreen.Y
            local barHeight = boxHeight
            local barWidth = 3

            espData.HealthBarBG.Position = Vector2.new(barX - 1, barY - 1)
            espData.HealthBarBG.Size = Vector2.new(barWidth + 2, barHeight + 2)
            espData.HealthBarBG.Visible = true

            local fillHeight = barHeight * healthPercent
            local healthColor = Color3.fromRGB(
                math.floor(255 * (1 - healthPercent)),
                math.floor(255 * healthPercent),
                50
            )

            espData.HealthBarFill.Position = Vector2.new(barX, barY + (barHeight - fillHeight))
            espData.HealthBarFill.Size = Vector2.new(barWidth, fillHeight)
            espData.HealthBarFill.Color = healthColor
            espData.HealthBarFill.Visible = true
        else
            espData.HealthBarBG.Visible = false
            espData.HealthBarFill.Visible = false
        end

        -- Head Dot ESP
        if Config.ESP.HeadDot then
            local headPos = WorldToScreen(head.Position)
            espData.HeadDot.Position = headPos
            espData.HeadDot.Color = espColor
            espData.HeadDot.Radius = math.clamp(1200 / rootDepth, 2, 8)
            espData.HeadDot.Visible = true
        else
            espData.HeadDot.Visible = false
        end

        -- Tracer ESP
        if Config.ESP.Tracer then
            local screenSize = Camera.ViewportSize
            espData.Tracer.From = Vector2.new(screenSize.X / 2, screenSize.Y)
            espData.Tracer.To = rootScreen
            espData.Tracer.Color = espColor
            espData.Tracer.Visible = true
        else
            espData.Tracer.Visible = false
        end

        -- Skeleton ESP
        if Config.ESP.Skeleton then
            local isR15 = character:FindFirstChild("UpperTorso") ~= nil
            local connections = isR15 and SkeletonConnections or SkeletonConnectionsR6

            for _, conn in ipairs(connections) do
                local part1 = character:FindFirstChild(conn[1])
                local part2 = character:FindFirstChild(conn[2])
                local lineName = conn[3]
                local line = espData.SkeletonLines[lineName]

                if line and part1 and part2 then
                    local p1Screen, p1OnScreen = WorldToScreen(part1.Position)
                    local p2Screen, p2OnScreen = WorldToScreen(part2.Position)

                    if p1OnScreen and p2OnScreen then
                        line.From = p1Screen
                        line.To = p2Screen
                        line.Color = Colors.AccentGlow
                        line.Visible = true
                    else
                        line.Visible = false
                    end
                elseif line then
                    line.Visible = false
                end
            end
        else
            for _, line in pairs(espData.SkeletonLines) do
                line.Visible = false
            end
        end
    end
end

local ChamsStorage = {}

local function ApplyChams(player)
    local character = GetCharacter(player)
    if not character then return end
    
    -- Удаляем старые чамы
    if ChamsStorage[player] then
        for _, highlight in pairs(ChamsStorage[player]) do
            pcall(function() highlight:Destroy() end)
        end
        ChamsStorage[player] = nil
    end

    if not Config.ESP.Chams or not Config.ESP.Enabled then return end
    if player == LocalPlayer then return end
    if Config.ESP.TeamCheck and IsSameTeam(player) then return end

    local highlight = Instance.new("Highlight")
    highlight.Name = "FleexiChams"
    highlight.Adornee = character
    highlight.FillColor = Config.ESP.ChamsColor
    highlight.FillTransparency = 0.3
    highlight.OutlineColor = Config.ESP.ChamsColor
    highlight.OutlineTransparency = 0
    highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    highlight.Parent = game:GetService("CoreGui")

    ChamsStorage[player] = {highlight}
end

local function RemoveChams(player)
    if ChamsStorage[player] then
        for _, highlight in pairs(ChamsStorage[player]) do
            pcall(function() highlight:Destroy() end)
        end
        ChamsStorage[player] = nil
    end
end

local function UpdateAllChams()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            if Config.ESP.Chams and Config.ESP.Enabled then
                ApplyChams(player)
            else
                RemoveChams(player)
            end
        end
    end
end

-- FOV Circle
FOVCircle = Drawing.new("Circle")
FOVCircle.Radius = Config.Aimbot.FOVSize
FOVCircle.Thickness = 1
FOVCircle.Color = Colors.AccentSecondary
FOVCircle.Filled = false
FOVCircle.Transparency = 0.6
FOVCircle.Visible = Config.Aimbot.FOVShow
FOVCircle.ZIndex = 10

local function GetClosestPlayerInFOV()
    local closestPlayer = nil
    local closestDistance = math.huge
    local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer then continue end
        if not IsAlive(player) then continue end
        if Config.ESP.TeamCheck and IsSameTeam(player) then continue end

        local character = GetCharacter(player)
        local targetPart = GetBodyPart(character, Config.Aimbot.Target)
        if not targetPart then continue end

        local screenPos, onScreen = WorldToScreen(targetPart.Position)
        if not onScreen then continue end

        local distFromCenter = (screenPos - screenCenter).Magnitude

        if distFromCenter <= Config.Aimbot.FOVSize and distFromCenter < closestDistance then
            closestDistance = distFromCenter
            closestPlayer = player
        end
    end

    return closestPlayer
end

local function AimbotUpdate()
    -- FOV Circle позиция
    local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    FOVCircle.Position = screenCenter
    FOVCircle.Radius = Config.Aimbot.FOVSize
    FOVCircle.Visible = Config.Aimbot.FOVShow and Config.Aimbot.Enabled

    if not Config.Aimbot.Enabled then return end

    -- Проверяем нажатие правой кнопки мыши (или левой для аимбота)
    local isAiming = UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)

    if not isAiming then return end

    local target = GetClosestPlayerInFOV()
    if not target then return end

    local character = GetCharacter(target)
    if not character then return end

    local targetPart = GetBodyPart(character, Config.Aimbot.Target)
    if not targetPart then return end

    if Config.Aimbot.SilentAim then
        -- Silent Aim: модифицируем Raycast (через метатаблицы если возможно)
        -- Для базовой реализации используем камерную наводку при стрельбе
        local targetCFrame = CFrame.new(Camera.CFrame.Position, targetPart.Position)
        Camera.CFrame = targetCFrame
    else
        -- Обычный аимбот с плавностью
        local targetPos = targetPart.Position
        local targetScreen, onScreen = WorldToScreen(targetPos)

        if onScreen then
            local currentCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
            local smoothFactor = Config.Aimbot.Smooth

            if smoothFactor <= 1 then
                -- Мгновенная наводка
                Camera.CFrame = CFrame.new(Camera.CFrame.Position, targetPos)
            else
                -- Плавная наводка
                local targetCFrame = CFrame.new(Camera.CFrame.Position, targetPos)
                local lerpAlpha = 1 / smoothFactor
                Camera.CFrame = Camera.CFrame:Lerp(targetCFrame, lerpAlpha)
            end
        end
    end
end

-- Silent Aim через namecall hook
pcall(function()
    if Config.Aimbot.SilentAim and getrawmetatable then
        local oldNamecall
        local mt = getrawmetatable(game)
        if mt then
            local oldIndex = mt.__namecall
            if setreadonly then setreadonly(mt, false) end

            mt.__namecall = newcclosure(function(self, ...)
                local method = getnamecallmethod()
                local args = {...}

                if Config.Aimbot.Enabled and Config.Aimbot.SilentAim then
                    if method == "FindPartOnRayWithIgnoreList" or method == "FindPartOnRay" then
                        local target = GetClosestPlayerInFOV()
                        if target then
                            local char = GetCharacter(target)
                            local part = char and GetBodyPart(char, Config.Aimbot.Target)
                            if part then
                                local origin = args[1] and typeof(args[1]) == "Ray" and args[1].Origin or Camera.CFrame.Position
                                local direction = (part.Position - origin).Unit * 1000
                                args[1] = Ray.new(origin, direction)
                                return oldIndex(self, unpack(args))
                            end
                        end
                    end
                end

                return oldIndex(self, ...)
            end)

            if setreadonly then setreadonly(mt, true) end
        end
    end
end)

-- CROSSHAIR СИСТЕМА (Пиксельное сердце)
-- ============================================================

local crosshairRotation = 0

-- Форма сердца в пиксельной сетке (базовая)
local function GetHeartPoints(size, density, rotation)
    local points = {}
    local step = math.max(0.05, 0.3 / math.max(density, 1))

    for t = 0, math.pi * 2, step do
        -- Параметрическое уравнение сердца
        local x = 16 * math.sin(t) ^ 3
        local y = -(13 * math.cos(t) - 5 * math.cos(2 * t) - 2 * math.cos(3 * t) - math.cos(4 * t))

        -- Масштабирование
        local scale = size / 16
        x = x * scale
        y = y * scale

        -- Вращение
        if rotation ~= 0 then
            local cosR = math.cos(rotation)
            local sinR = math.sin(rotation)
            local rx = x * cosR - y * sinR
            local ry = x * sinR + y * cosR
            x = rx
            y = ry
        end

        table.insert(points, {x = x, y = y})
    end

    return points
end

local function UpdateCrosshair()
    -- Очистить старые точки
    for _, dot in ipairs(CrosshairDots) do
        CleanDrawing(dot)
    end
    CrosshairDots = {}

    if not Config.Crosshair.Enabled then return end

    local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    -- Обновить вращение
    if Config.Crosshair.Rotate then
        crosshairRotation = crosshairRotation + math.rad(Config.Crosshair.RotateSpeed)
        if crosshairRotation > math.pi * 2 then
            crosshairRotation = crosshairRotation - math.pi * 2
        end
    else
        crosshairRotation = 0
    end

    local points = GetHeartPoints(Config.Crosshair.Size, Config.Crosshair.PixelDensity, crosshairRotation)

    for _, point in ipairs(points) do
        local dot = Drawing.new("Circle")
        dot.Position = Vector2.new(screenCenter.X + point.x, screenCenter.Y + point.y)
        dot.Radius = Config.Crosshair.Thickness
        dot.Filled = true
        dot.Color = Colors.AccentSecondary
        dot.Visible = true
        dot.ZIndex = 20
        dot.Transparency = 1
        table.insert(CrosshairDots, dot)
    end
end

getgenv().FleexiCleanup = function()
    -- Отключить все соединения
    if ESPConnection then ESPConnection:Disconnect() end
    if ChamsConnection then ChamsConnection:Disconnect() end

    for _, conn in ipairs(InputConnections) do
        pcall(function() conn:Disconnect() end)
    end
    InputConnections = {}

    -- Удалить все ESP Drawing объекты
    for player, _ in pairs(ESPObjects) do
        RemoveESPForPlayer(player)
    end

    -- Удалить все Chams
    for player, _ in pairs(ChamsStorage) do
        RemoveChams(player)
    end

    -- Удалить Crosshair
    for _, dot in ipairs(CrosshairDots) do
        CleanDrawing(dot)
    end
    CrosshairDots = {}

    -- Удалить FOV Circle
    CleanDrawing(FOVCircle)

    -- Удалить GUI
    pcall(function() ScreenGui:Destroy() end)

    getgenv().FleexiState = nil
    getgenv().FleexiCleanup = nil
end
