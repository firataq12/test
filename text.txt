local function W2S(pos)
    local s, o = Camera:WorldToViewportPoint(pos)
    return Vector2.new(s.X, s.Y), o, s.Z
end

local function Dist(a, b) return (a - b).Magnitude end

local function Alive(p)
    local ch = p and p.Character
    if not ch then return false end
    local h = ch:FindFirstChildOfClass("Humanoid")
    local r = ch:FindFirstChild("HumanoidRootPart")
    return h and r and h.Health > 0
end

local function SameTeam(p)
    if not CFG.ESP_TeamCheck then return false end
    return LP.Team and p.Team and LP.Team == p.Team
end

local function SafeRemove(d)
    if d then pcall(function() d:Remove() end) end
end

local function MakeCorner(p, r)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, r or 6)
    c.Parent = p
    return c
end

local function MakeStroke(p, col, t)
    local s = Instance.new("UIStroke")
    s.Color = col or C.Border
    s.Thickness = t or 1
    s.Parent = p
    return s
end

local function MakePad(p, t, b, l, r)
    local pd = Instance.new("UIPadding")
    pd.PaddingTop = UDim.new(0, t or 0)
    pd.PaddingBottom = UDim.new(0, b or 0)
    pd.PaddingLeft = UDim.new(0, l or 0)
    pd.PaddingRight = UDim.new(0, r or 0)
    pd.Parent = p
    return pd
end

    page.AutomaticSize = Enum.AutomaticSize.Y

    page.BorderSizePixel = 0

    local lay = Instance.new("UIListLayout")

    lay.HorizontalAlignment = Enum.HorizontalAlignment.Center
    lay.SortOrder = Enum.SortOrder.LayoutOrder
    lay.Parent = page

    return page

local function SwitchTo(name)
    ActiveTab = name
    for n, d in pairs(TabData) do
        local active = (n == name)
        d.page.Visible = active
        d.btn.BackgroundColor3 = active and C.Red1 or C.BG3
        d.btn.TextColor3 = active and C.White or C.TxtG
    end
    RightPanel.CanvasPosition = Vector2.new(0, 0)
end

local function AddTab(name, order)
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(1, 0, 0, 30)
    btn.BackgroundColor3 = C.BG3
    btn.Text = name
    btn.TextColor3 = C.TxtG
    btn.TextSize = 11
    btn.Font = Enum.Font.GothamSemibold
    btn.BorderSizePixel = 0
    btn.AutoButtonColor = false
    btn.LayoutOrder = order
    btn.ZIndex = 102
    btn.Parent = LeftPanel
    MakeCorner(btn, 5)

    local page = MakeTabPage()
    page.LayoutOrder = order

    TabData[name] = {btn = btn, page = page}

    btn.MouseButton1Click:Connect(function()
        SwitchTo(name)
    end)

    -- Dokunmatik destek
    btn.TouchTap:Connect(function()
        SwitchTo(name)
    end)

    return page
end

local function AddSlider(page, text, mn, mx, default, ord, cb)
    local f = Instance.new("Frame")
    f.Size = UDim2.new(1, 0, 0, 44)
    f.BackgroundColor3 = C.BG3
    f.BorderSizePixel = 0
    f.LayoutOrder = ord
    f.ZIndex = 102
    f.Parent = page
    MakeCorner(f, 5)

    local l = Instance.new("TextLabel")
    l.Size = UDim2.new(1, -55, 0, 18)
    l.Position = UDim2.new(0, 8, 0, 2)
    l.BackgroundTransparency = 1
    l.Text = text
    l.TextColor3 = C.TxtW
    l.TextSize = 10
    l.Font = Enum.Font.Gotham
    l.TextXAlignment = Enum.TextXAlignment.Left
    l.TextTruncate = Enum.TextTruncate.AtEnd
    l.ZIndex = 103
    l.Parent = f

    local vl = Instance.new("TextLabel")
    vl.Size = UDim2.new(0, 45, 0, 18)
    vl.Position = UDim2.new(1, -50, 0, 2)
    vl.BackgroundTransparency = 1
    vl.Text = tostring(math.floor(default))
    vl.TextColor3 = C.Red4
    vl.TextSize = 11
    vl.Font = Enum.Font.GothamBold
    vl.TextXAlignment = Enum.TextXAlignment.Right
    vl.ZIndex = 103
    vl.Parent = f

    local track = Instance.new("Frame")
    track.Size = UDim2.new(1, -16, 0, 6)
    track.Position = UDim2.new(0, 8, 0, 28)
    track.BackgroundColor3 = C.Off
    track.BorderSizePixel = 0
    track.ZIndex = 103
    track.Parent = f
    MakeCorner(track, 3)

    local pct = math.clamp((default - mn) / (mx - mn), 0, 1)
    local fill = Instance.new("Frame")
    fill.Size = UDim2.new(pct, 0, 1, 0)
    fill.BackgroundColor3 = C.Red1
    fill.BorderSizePixel = 0
    fill.ZIndex = 104
    fill.Parent = track
    MakeCorner(fill, 3)

    local knob = Instance.new("Frame")
    knob.Size = UDim2.new(0, 14, 0, 14)
    knob.Position = UDim2.new(pct, -7, 0.5, -7)
    knob.BackgroundColor3 = C.Red3
    knob.BorderSizePixel = 0
    knob.ZIndex = 105
    knob.Parent = track
    MakeCorner(knob, 7)

    local curVal = default
    local dragging = false
    local maxVal = mx

    local function setVal(v)
        curVal = math.clamp(math.floor(v), mn, maxVal)
        local p2 = math.clamp((curVal - mn) / (maxVal - mn), 0, 1)
        fill.Size = UDim2.new(p2, 0, 1, 0)
        knob.Position = UDim2.new(p2, -7, 0.5, -7)
        vl.Text = tostring(curVal)
        if cb then cb(curVal) end
    end

    local function handleInput(x)
        local tAbs = track.AbsolutePosition.X
        local tSize = track.AbsoluteSize.X
        if tSize <= 0 then return end
        local p3 = math.clamp((x - tAbs) / tSize, 0, 1)
        setVal(mn + (maxVal - mn) * p3)
    end

    -- Dokunmatik + fare desteği
    local inputBtn = Instance.new("TextButton")
    inputBtn.Size = UDim2.new(1, 0, 0, 22)
    inputBtn.Position = UDim2.new(0, 0, 0, 22)
    inputBtn.BackgroundTransparency = 1
    inputBtn.Text = ""
    inputBtn.ZIndex = 106
    inputBtn.Parent = f

    inputBtn.MouseButton1Down:Connect(function()
        dragging = true
        handleInput(game:GetService("Players").LocalPlayer:GetMouse().X)
    end)

    inputBtn.TouchPan:Connect(function(positions)
        if #positions > 0 then
            handleInput(positions[1].Position.X)
        end
    end)

    inputBtn.TouchTap:Connect(function(positions)
        if #positions > 0 then
            handleInput(positions[1].X)
        end
    end)

    table.insert(Connections, UIS.InputChanged:Connect(function(inp)
        if dragging then
            if inp.UserInputType == Enum.UserInputType.MouseMovement then
                handleInput(inp.Position.X)
            elseif inp.UserInputType == Enum.UserInputType.Touch then
                handleInput(inp.Position.X)
            end
        end
    end))

    table.insert(Connections, UIS.InputEnded:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end))

    return {
        Set = function(v) setVal(v) end,
        Get = function() return curVal end,
        SetMax = function(m) maxVal = m end
    }
end

local function AddSelector(page, text, opts, defIdx, ord, cb)

    local selIdx = defIdx or 1

    local bContainer = Instance.new("Frame")
    bContainer.Size = UDim2.new(1, -82, 1, -4)
    bContainer.Position = UDim2.new(0, 78, 0, 2)
    bContainer.BackgroundTransparency = 1
    bContainer.ZIndex = 103
    bContainer.Parent = f

    local bLay = Instance.new("UIListLayout")
    bLay.FillDirection = Enum.FillDirection.Horizontal
    bLay.Padding = UDim.new(0, 2)
    bLay.HorizontalAlignment = Enum.HorizontalAlignment.Left
    bLay.Parent = bContainer

    local bw = math.floor((580 - 130 - (#opts) * 2) / #opts)
    bw = math.min(bw, 90)

    for i, opt in ipairs(opts) do
        local b = Instance.new("TextButton")
        b.Size = UDim2.new(0, bw, 1, 0)
        b.BackgroundColor3 = (i == selIdx) and C.Red1 or C.RedDark
        b.Text = opt
        b.TextColor3 = (i == selIdx) and C.White or C.TxtM
        b.TextSize = 9
        b.Font = Enum.Font.GothamSemibold
        b.BorderSizePixel = 0
        b.AutoButtonColor = false
        b.ZIndex = 104
        b.TextTruncate = Enum.TextTruncate.AtEnd
        b.Parent = bContainer
        MakeCorner(b, 4)

        btns[i] = b

        local function sel()
            selIdx = i
            for j, bb in ipairs(btns) do
                bb.BackgroundColor3 = (j == i) and C.Red1 or C.RedDark
                bb.TextColor3 = (j == i) and C.White or C.TxtM
            end
            if cb then cb(i) end

        b.TouchTap:Connect(sel)
    end

        Get = function() return selIdx end,
        Set = function(i)
            selIdx = i
            for j, bb in ipairs(btns) do
                bb.BackgroundColor3 = (j == i) and C.Red1 or C.RedDark
                bb.TextColor3 = (j == i) and C.White or C.TxtM
            end
        end
    }
end

local function DestroyESP(player)
    local d = ESPCache[player]
    if not d then return end

    SafeRemove(d.box)
    SafeRemove(d.boxOut)
    SafeRemove(d.name)
    SafeRemove(d.dist)
    SafeRemove(d.head)
    SafeRemove(d.tracer)
    SafeRemove(d.hpBG)
    SafeRemove(d.hpFill)

    if d.bones then
        for _, ln in ipairs(d.bones) do
            SafeRemove(ln)
        end
    end

    ESPCache[player] = nil
end

    end
end

local function RunESP()
    if not CFG.ESP_On then
        for _, d in pairs(ESPCache) do
            HideAll(d)
        end
        return
    end

    local myChar = LP.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end

    for _, player in ipairs(Players:GetPlayers()) do
        if player == LP then continue end

        if not ESPCache[player] then
            MakeESP(player)
        end

        local d = ESPCache[player]
        if not d then continue end

        if not Alive(player) then
            HideAll(d)
            continue
        end

        if CFG.ESP_TeamCheck and SameTeam(player) then
            HideAll(d)
            continue
        end

        local ch = player.Character
        local root = ch:FindFirstChild("HumanoidRootPart")
        local head2 = ch:FindFirstChild("Head")
        local hum = ch:FindFirstChildOfClass("Humanoid")

        if not (root and head2 and hum) then
            HideAll(d)
            continue
        end

        local dist = Dist(myRoot.Position, root.Position)
        if dist > CFG.ESP_MaxDist then
            HideAll(d)
            continue
        end

        local rootScr, rootOn, rootZ = W2S(root.Position)
        local headTop, headOn = W2S(head2.Position + Vector3.new(0, 0.6, 0))
        local feetScr = W2S(root.Position - Vector3.new(0, 3, 0))

        if not rootOn then
            HideAll(d)
            continue
        end

        local isFriend = SameTeam(player)
        local col = isFriend and C.Friend or C.Enemy

        local bH = math.abs(headTop.Y - feetScr.Y)
        local bW = bH * 0.55
        local bX = rootScr.X - bW / 2
        local bY = headTop.Y

        -- BOX
        if CFG.ESP_Box then
            d.boxOut.Size = Vector2.new(bW, bH)
            d.boxOut.Position = Vector2.new(bX, bY)
            d.boxOut.Visible = true

            d.box.Size = Vector2.new(bW, bH)
            d.box.Position = Vector2.new(bX, bY)
            d.box.Color = col
            d.box.Visible = true
        else
            d.box.Visible = false
            d.boxOut.Visible = false
        end

        -- NAME
        if CFG.ESP_Name then
            d.name.Position = Vector2.new(rootScr.X, bY - 16)
            d.name.Text = player.DisplayName
            d.name.Color = col
            d.name.Visible = true
        else
            d.name.Visible = false
        end

        -- DISTANCE
        if CFG.ESP_Distance then
            d.dist.Position = Vector2.new(rootScr.X, feetScr.Y + 3)
            d.dist.Text = math.floor(dist) .. "м"
            d.dist.Color = C.TxtG
            d.dist.Visible = true
        else
            d.dist.Visible = false
        end

        -- HEALTH BAR
        if CFG.ESP_HealthBar then
            local hp = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
            local hbX = bX - 5
            local hbH = bH

            d.hpBG.Position = Vector2.new(hbX - 1, bY - 1)
            d.hpBG.Size = Vector2.new(4, hbH + 2)
            d.hpBG.Visible = true

            local fH = hbH * hp
            d.hpFill.Position = Vector2.new(hbX, bY + (hbH - fH))
            d.hpFill.Size = Vector2.new(2, fH)
            d.hpFill.Color = Color3.fromRGB(255 * (1 - hp), 255 * hp, 40)
            d.hpFill.Visible = true
        else
            d.hpBG.Visible = false
            d.hpFill.Visible = false
        end

        -- HEAD DOT
        if CFG.ESP_HeadDot then
            local hScr = W2S(head2.Position)
            d.head.Position = hScr
            d.head.Radius = math.clamp(800 / rootZ, 2, 6)
            d.head.Color = col
            d.head.Visible = true
        else
            d.head.Visible = false
        end

        -- TRACER
        if CFG.ESP_Tracer then
            local vs = Camera.ViewportSize
            d.tracer.From = Vector2.new(vs.X / 2, vs.Y)
            d.tracer.To = rootScr
            d.tracer.Color = col
            d.tracer.Visible = true
        else
            d.tracer.Visible = false
        end

        -- SKELETON
        if CFG.ESP_Skeleton then
            local isR15 = ch:FindFirstChild("UpperTorso") ~= nil
            local boneList = isR15 and SkeletonBonesR15 or SkeletonBonesR6

            for i, bone in ipairs(boneList) do
                local ln = d.bones[i]
                if not ln then continue end

                local p1 = ch:FindFirstChild(bone[1])
                local p2 = ch:FindFirstChild(bone[2])

                if p1 and p2 then
                    local s1, o1 = W2S(p1.Position)
                    local s2, o2 = W2S(p2.Position)

                    if o1 and o2 then
                        ln.From = s1
                        ln.To = s2
                        ln.Color = C.Red4
                        ln.Visible = true
                    else
                        ln.Visible = false
                    end
                else
                    ln.Visible = false
                end
            end

            -- Kullanılmayan kemik çizgilerini gizle
            for i = #boneList + 1, 14 do
                if d.bones[i] then
                    d.bones[i].Visible = false
                end
            end
        else
            for _, ln in ipairs(d.bones) do
                ln.Visible = false
            end
        end
    end
end

local function ApplyChams()

local AimTarget = nil

local function GetClosestInFOV()
    local best = nil
    local bestDist = math.huge
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    for _, player in ipairs(Players:GetPlayers()) do
        if player == LP then continue end
        if not Alive(player) then continue end
        if CFG.ESP_TeamCheck and SameTeam(player) then continue end

        local ch = player.Character
        local partName = TargetParts[CFG.AIM_TargetIdx] or "Head"
        local part = ch and ch:FindFirstChild(partName)
        if not part then
            part = ch and ch:FindFirstChild("HumanoidRootPart")
        end
        if not part then continue end

        local scr, onScr = W2S(part.Position)
        if not onScr then continue end

        local d2 = (scr - center).Magnitude
        if d2 <= CFG.AIM_FOVSize and d2 < bestDist then
            bestDist = d2
            best = player
        end
    end

    return best
end

local function RunAimbot()
    -- FOV Circle güncelle
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    FOVCircle.Position = center
    FOVCircle.Radius = CFG.AIM_FOVSize
    FOVCircle.Visible = CFG.AIM_On and CFG.AIM_FOVShow

    if not CFG.AIM_On then return end

    -- Mobilde dokunma kontrolü: ekranın sağ tarafına dokunma = nişan alma
    local aiming = false

    -- Mouse button (emülatör) veya dokunma
    if UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
        aiming = true
    end

    -- Mobil: aktif dokunuşları kontrol et
    if not aiming then
        local touches = UIS:GetMouseLocation()
        -- Basit yöntem: aimbot her zaman aktif hedefi bulsun (mobil mod)
        -- Kullanıcı ateş butonuna bastığında silent aim devreye girer
        if CFG.AIM_Silent then
            aiming = true -- Silent aim sürekli hedef takip eder
        end
    end

    -- Dokunmatik ekran desteği: tüm mevcut dokunuşları kontrol et
    if not aiming then
        local touchInputs = UIS:GetGamepadState(Enum.UserInputType.Touch)
        -- Alternatif: basit approach - aiming butonu olmadan çalışır
    end

    local target = GetClosestInFOV()
    if not target then return end

    local ch = target.Character
    if not ch then return end

    local partName = TargetParts[CFG.AIM_TargetIdx] or "Head"
    local part = ch:FindFirstChild(partName) or ch:FindFirstChild("HumanoidRootPart")
    if not part then return end

    if CFG.AIM_Silent then
        -- Silent Aim: sürekli hedef kilitler ama ekran oynamaz
        -- Raycast hook ile çalışır
        AimTarget = part
    end

    if aiming and not CFG.AIM_Silent then
        local targetPos = part.Position
        if CFG.AIM_Smooth <= 1 then
            Camera.CFrame = CFrame.new(Camera.CFrame.Position, targetPos)
        else
            local targetCF = CFrame.new(Camera.CFrame.Position, targetPos)
            Camera.CFrame = Camera.CFrame:Lerp(targetCF, 1 / CFG.AIM_Smooth)
        end
    end
end

-- Silent Aim Hook (Delta/Codex uyumlu)
pcall(function()
    local mt = getrawmetatable(game)
    if not mt then return end

    local oldNamecall = mt.__namecall
    if setreadonly then setreadonly(mt, false) end

    mt.__namecall = newcclosure(function(self, ...)
        local method = getnamecallmethod()

        if CFG.AIM_On and CFG.AIM_Silent and AimTarget then
            if method == "FindPartOnRayWithIgnoreList" or
               method == "FindPartOnRay" or
               method == "Raycast" or
               method == "FindPartOnRayWithWhitelist" then

                local args = {...}
                if typeof(args[1]) == "Ray" then
                    local origin = args[1].Origin
                    local dir = (AimTarget.Position - origin).Unit * 1000
                    args[1] = Ray.new(origin, dir)
                    return oldNamecall(self, unpack(args))
                elseif typeof(self) == "Instance" and self:IsA("WorldRoot") and method == "Raycast" then
                    local origin = args[1]
                    if typeof(origin) == "Vector3" then
                        local dir = (AimTarget.Position - origin).Unit * 1000
                        args[2] = dir
                        return oldNamecall(self, unpack(args))
                    end
                end
            end
        end

        return oldNamecall(self, ...)
    end)

    if setreadonly then setreadonly(mt, true) end
end)

local function HeartPoints(size, density, angle)
    local pts = {}
    local step = math.max(0.03, 0.25 / math.max(density, 1))

    for t = 0, math.pi * 2, step do
        local x = 16 * math.sin(t) * math.sin(t) * math.sin(t)
        local y = -(13 * math.cos(t) - 5 * math.cos(2 * t) - 2 * math.cos(3 * t) - math.cos(4 * t))

        local s = size / 16
        x = x * s
        y = y * s

        if angle ~= 0 then
            local ca = math.cos(angle)
            local sa = math.sin(angle)
            local rx = x * ca - y * sa
            local ry = x * sa + y * ca
            x = rx
            y = ry
        end

        table.insert(pts, Vector2.new(x, y))
    end

    return pts
end

local function RunCrosshair()
    ClearCrosshair()

    if not CFG.CROSS_On then return end

    local cx = Camera.ViewportSize.X / 2
    local cy = Camera.ViewportSize.Y / 2

    if CFG.CROSS_Rotate then
        CrosshairAngle = CrosshairAngle + math.rad(CFG.CROSS_RotSpd * 0.5)
        if CrosshairAngle > math.pi * 2 then
            CrosshairAngle = CrosshairAngle - math.pi * 2
        end
    else
        CrosshairAngle = 0
    end

    local points = HeartPoints(CFG.CROSS_Size, CFG.CROSS_Density, CrosshairAngle)

    for _, pt in ipairs(points) do
        local dot = Drawing.new("Circle")
        dot.Position = Vector2.new(cx + pt.X, cy + pt.Y)
        dot.Radius = CFG.CROSS_Thick
        dot.Filled = true
        dot.Color = C.Red3
        dot.Visible = true
        dot.ZIndex = 20
        dot.Transparency = 1
        table.insert(CrosshairCache, dot)
    end
end

local frameCount = 0

local MainLoop = RunService.RenderStepped:Connect(function()
    frameCount = frameCount + 1

    -- ESP her frame
    RunESP()

    -- Aimbot her frame
    RunAimbot()
    RunAimbotMobile()

    -- Crosshair her frame
    RunCrosshair()

    -- Chams her 60 frame (~1 saniye)
    if frameCount % 60 == 0 then
        ApplyChams()
    end
end)

table.insert(Connections, MainLoop)

-- ============================================================
-- OYUNCU BAĞLANTILARI
-- ============================================================
table.insert(Connections, Players.PlayerAdded:Connect(function(p)
    task.wait(1)
    MakeESP(p)
    p.CharacterAdded:Connect(function()
        task.wait(0.5)
        if CFG.ESP_Chams and CFG.ESP_On then
            ApplyChams()
        end
    end)
end))

table.insert(Connections, Players.PlayerRemoving:Connect(function(p)
    DestroyESP(p)
    if ChamsCache[p] then
        pcall(function() ChamsCache[p]:Destroy() end)
        ChamsCache[p] = nil
    end
end))

-- Mevcut oyuncular için ESP oluştur
for _, p in ipairs(Players:GetPlayers()) do
    if p ~= LP then
        MakeESP(p)
        pcall(function()
            p.CharacterAdded:Connect(function()
                task.wait(0.5)
                if CFG.ESP_Chams and CFG.ESP_On then
                    ApplyChams()
                end
            end)
        end)
    end
end

-- ============================================================
-- TEMİZLEME FONKSİYONU
-- ============================================================
getgenv().FleexiCleanup = function()
    -- Bağlantıları kes
    for _, conn in ipairs(Connections) do
        pcall(function() conn:Disconnect() end)
    end
    Connections = {}

    -- ESP temizle
    for p, _ in pairs(ESPCache) do
        DestroyESP(p)
    end
    ESPCache = {}

    -- Chams temizle
    for p, hl in pairs(ChamsCache) do
        pcall(function() hl:Destroy() end)
    end
    ChamsCache = {}

    -- Crosshair temizle
    ClearCrosshair()

    -- FOV Circle temizle
    SafeRemove(FOVCircle)

    -- GUI sil
    pcall(function() Gui:Destroy() end)

    getgenv().FleexiCleanup = nil
end
