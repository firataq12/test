local function СделатьДрагабельным(frame)
    local тащим   = false
    local началоТаска = nil
    local начПоз  = nil

    frame.InputBegan:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.Touch
        or inp.UserInputType == Enum.UserInputType.MouseButton1 UDim2.new(рацио, 0, 1, 0)
        кнобик.Position = UDim2.new(рацио, 0, 0.5, 0)
        знач.Text       = tostring(рез)
        коллбек(рез)
    end

    -- МОБИЛЬНЫЙ + ПК: через хитбокс
    хитбокс.InputBegan:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1
        or inp.UserInputType == Enum.UserInputType.Touch then
            тащим = true
            обновитьПоX(inp.Position.X)
        end
    end)

    хитбокс.InputEnded:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1
        or inp.UserInputType == Enum.UserInputType.Touch then
            тащим = false
        end
    end)

    -- Движение мыши/пальца
    UserInputService.InputChanged:Connect(function(inp)
        if not тащим then return end
        if inp.UserInputType == Enum.UserInputType.MouseMovement
        or inp.UserInputType == Enum.UserInputType.Touch then
            обновитьПоX(inp.Position.X)
        end
    end)

    -- Также через трек напрямую
    трекФон.InputBegan:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1
        or inp.UserInputType == Enum.UserInputType.Touch then
            тащим = true
            обновитьПоX(inp.Position.X)
        end
    end)

    UserInputService.InputEnded:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1
        or inp.UserInputType == Enum.UserInputType.Touch then
            тащим = false
        end
    end)
end

local function ПолучитьГраницы(персонаж)
    local хрп = персонаж:FindFirstChild("HumanoidRootPart")
    if not хрп then return nil end

    local сп = Camera:WorldToViewportPoint(хрп.Position)
    if сп.Z <= 0 then return nil end

    local ок, цф, размер = pcall(function()
        return персонаж:GetBoundingBox()
    end)
    if not ок or not цф or not размер then return nil end

    local хх = размер.X * 0.5
    local хy = размер.Y * 0.5
    local хz = размер.Z * 0.5

    local углы = {
        Vector3.new(-хх,-хy,-хz), Vector3.new( хх,-хy,-хz),
        Vector3.new(-хх, хy,-хz), Vector3.new( хх, хy,-хz),
        Vector3.new(-хх,-хy, хz), Vector3.new( хх,-хy, хz),
        Vector3.new(-хх, хy, хz), Vector3.new( хх, хy, хz),
    }

    local минX, минY =  math.huge,  math.huge
    local максX, максY = -math.huge, -math.huge
    local счёт = 0

    for _, смещ in ipairs(углы) do
        local мп = цф:PointToWorldSpace(смещ)
        local сп2 = Camera:WorldToViewportPoint(мп)
        if сп2.Z > 0 then
            счёт = счёт + 1
            if сп2.X < минX then минX = сп2.X end
            if сп2.X > максX then максX = сп2.X end
            if сп2.Y < минY then минY = сп2.Y end
            if сп2.Y > максY then максY = сп2.Y end
        end
    end

    if счёт == 0 then return nil end

    local Ш = максX - минX
    local В = максY - минY
    if Ш < 4 or В < 4 or Ш > 3000 or В > 3000 then return nil end

    return минX - 2, минY - 2, максX + 2, максY + 2, Ш + 4, В + 4
end

local function ОбновитьЕСП()
    for _, игр in ipairs(Players:GetPlayers()) do
        if игр == LocalPlayer then continue end
        if not ЕСП_Кэш[игр] then СоздатьЕСП(игр) end

        local о    = ЕСП_Кэш[игр]
        local перс = игр.Character
        local гум  = перс and перс:FindFirstChildOfClass("Humanoid")
        local хрп  = перс and перс:FindFirstChild("HumanoidRootPart")

        local однаКоманда = Config.ESP_КомандаЧек
            and игр.Team ~= nil
            and LocalPlayer.Team ~= nil
            and игр.Team == LocalPlayer.Team

        if not Config.ESP_Включен or not перс or not гум
           or гум.Health <= 0 or not хрп or однаКоманда then
            СкрытьЕСП(о)
            continue
        end

        -- Чамс
        if Config.Чамс_Включен then
            о.Чамс.Adornee        = перс
            о.Чамс.Enabled        = true
            о.Чамс.FillTransparency = Config.Чамс_Прозрачность
        else
            о.Чамс.Enabled  = false
            о.Чамс.Adornee  = nil
        end

        local минX, минY, максX, максY, Ш, В = ПолучитьГраницы(перс)

        if not минX then
            о.Бокс.Visible    = false
            о.Имя.Visible     = false
            о.Дист.Visible    = false
            о.ХПФон.Visible   = false
            о.Трассер.Visible = false
            continue
        end

        local цX = минX + Ш * 0.5

        if Config.ESP_Боксы then
            о.Бокс.Visible  = true
            о.Бокс.Position = UDim2.new(0, минX, 0, минY)
            о.Бокс.Size     = UDim2.new(0, Ш, 0, В)
        else
            о.Бокс.Visible = false
        end

        if Config.ESP_Имена then
            local имя = (игр.DisplayName ~= "" and игр.DisplayName) or игр.Name
            о.Имя.Visible  = true
            о.Имя.Text     = имя
            о.Имя.Position = UDim2.new(0, цX - 80, 0, минY - 18)
        else
            о.Имя.Visible = false
        end

        if Config.ESP_Дистанция then
            local дист = math.floor((Camera.CFrame.Position - хрп.Position).Magnitude)
            о.Дист.Visible  = true
            о.Дист.Text     = дист .. " м"
            о.Дист.Position = UDim2.new(0, цX - 50, 0, максY + 2)
        else
            о.Дист.Visible = false
        end

        if Config.ESP_Здоровье then
            local проц = math.clamp(гум.Health / math.max(гум.MaxHealth, 1), 0, 1)
            local барШ  = 4
            о.ХПФон.Visible  = true
            о.ХПФон.Position = UDim2.new(0, минX - барШ - 2, 0, минY)
            о.ХПФон.Size     = UDim2.new(0, барШ, 0, В)
            о.ХПЗаливка.BackgroundColor3 = Color3.fromRGB(
                math.floor(255 * (1 - проц)),
                math.floor(255 * проц), 0
            )
            о.ХПЗаливка.Size     = UDim2.new(1, 0, проц, 0)
            о.ХПЗаливка.Position = UDim2.new(0, 0, 1 - проц, 0)
        else
            о.ХПФон.Visible = false
        end

        if Config.ESP_Трассеры then
            local вп     = Camera.ViewportSize
            local нX, нY = вп.X * 0.5, вп.Y
            local кX, кY = цX, максY
            local дX, дY = кX - нX, кY - нY
            local длина  = math.sqrt(дX*дX + дY*дY)
            local угол   = math.deg(math.atan2(дY, дX))

            о.Трассер.Visible  = true
            о.Трассер.Size     = UDim2.new(0, длина, 0, 1.5)
            о.Трассер.Position = UDim2.new(0, (нX+кX)*0.5, 0, (нY+кY)*0.5)
            о.Трассер.Rotation = угол
        else
            о.Трассер.Visible = false
        end
    end
end

-- ============================================================
-- ПРОВЕРКА СТЕН (RAYCAST)
-- ============================================================
local function ЕстьПрямаяВидимость(откуда, куда, игнорПерс)
    local направл = куда - откуда
    local расст   = направл.Magnitude
    if расст < 0.1 then return true end

    local парамс = RaycastParams.new()
    парамс.FilterType = Enum.RaycastFilterType.Exclude

    local список = {}
    if LocalPlayer.Character then
        table.insert(список, LocalPlayer.Character)
    end
    if игнорПерс then
        table.insert(список, игнорПерс)
    end
    парамс.FilterDescendantsInstances = список

    local рез = workspace:Raycast(откуда, направл.Unit * расст, парамс)
    return рез == nil
end

local аимАктивен = false

-- Мобильный: отслеживаем нажатие на правую часть экрана
local мобАимАктивен = false
local мобТач = nil

UserInputService.TouchStarted:Connect(function(инп, гпе)
    if гпе then return end
    local вп = Camera.ViewportSize
    -- Правая половина экрана = аим (мобильный аналог ПКМ)
    if инп.Position.X > вп.X * 0.55 then
        мобАимАктивен = true
        мобТач = инп
    end
end)

UserInputService.TouchEnded:Connect(function(инп)
    if мобТач and инп == мобТач then
        мобАимАктивен = false
        мобТач = nil
    end
end)

-- ПК поддержка
UserInputService.InputBegan:Connect(function(инп, гпе)
    if гпе then return end
    if инп.UserInputType == Enum.UserInputType.MouseButton2 then
        аимАктивен = true
    end
end)

UserInputService.InputEnded:Connect(function(инп)
    if инп.UserInputType == Enum.UserInputType.MouseButton2 then
        аимАктивен = false
    end
end)

local function ПолучитьЦель()
    local лучшая, лучшДист = nil, Config.Аимбот_ФОВ
    local вп      = Camera.ViewportSize
    local центр   = Vector2.new(вп.X * 0.5, вп.Y * 0.5)
    local камПоз  = Camera.CFrame.Position

    for _, игр in ipairs(Players:GetPlayers()) do
        if игр == LocalPlayer then continue end

        if Config.Аимбот_КомандаЧек
           and игр.Team ~= nil
           and LocalPlayer.Team ~= nil
           and игр.Team == LocalPlayer.Team then
            continue
        end

        local перс = игр.Character
        local гум  = перс and перс:FindFirstChildOfClass("Humanoid")
        local часть = перс and (
            перс:FindFirstChild("Head") or
            перс:FindFirstChild("HumanoidRootPart")
        )

        if not перс or not гум or гум.Health <= 0 or not часть then continue end

        -- Дистанция в мире
        local мирДист = (камПоз - часть.Position).Magnitude
        if мирДист > Config.Аимбот_МаксДист then continue end

        -- Позиция на экране
        local сп = Camera:WorldToViewportPoint(часть.Position)
        if сп.Z <= 0 then continue end

        local экрДист = (Vector2.new(сп.X, сп.Y) - центр).Magnitude
        if экрДист >= лучшДист then continue end

        -- Проверка стен
        if Config.Аимбот_СтенаЧек then
            if not ЕстьПрямаяВидимость(камПоз, часть.Position, перс) then
                continue
            end
        end

        лучшДист = экрДист
        лучшая   = часть
    end

    return лучшая
end

-- ============================================================
-- ОСНОВНОЙ АИМБОТ МЕТОД ДЛЯ DELTA/CODEX
-- Используем Camera CFrame напрямую - работает на мобильных
-- ============================================================
local function ВыполнитьАимбот(dt)
    local активен = аимАктивен or мобАимАктивен
    if not Config.Аимбот_Включен or not активен then
        -- Восстанавливаем камеру
        if Camera.CameraType == Enum.CameraType.Scriptable then
            Camera.CameraType = Enum.CameraType.Custom
        end
        return
    end

    local цель = ПолучитьЦель()
    if not цель then return end

    -- Предсказание
    local целПоз = цель.Position
    local хрп = цель.Parent and цель.Parent:FindFirstChild("HumanoidRootPart")
    if хрп and Config.Аимбот_Предсказ > 0 then
        целПоз = целПоз + хрп.AssemblyLinearVelocity * (Config.Аимбот_Предсказ * 0.016)
    end

    -- Устанавливаем тип камеры
    Camera.CameraType = Enum.CameraType.Scriptable

    local начало  = Camera.CFrame.Position
    local цельЦФ  = CFrame.new(начало, целПоз)

    if Config.Аимбот_Плавность <= 1 then
        -- 0ms мгновенный
        Camera.CFrame = цельЦФ
    else
        -- Плавный
        local альфа = math.clamp(dt * (30 / Config.Аимбот_Плавность), 0, 1)
        Camera.CFrame = Camera.CFrame:Lerp(цельЦФ, альфа)
    end
end

local углыСердца = 0

RunService.RenderStepped:Connect(function(дт)

    -- ESP
    ОбновитьЕСП()

    -- FOV Круг
    if Config.Аимбот_Включен and Config.Аимбот_ПоказФОВ then
        local вп   = Camera.ViewportSize
        local р    = Config.Аимбот_ФОВ
        ФОВ_Круг.Visible  = true
        ФОВ_Круг.Size     = UDim2.new(0, р * 2, 0, р * 2)
        ФОВ_Круг.Position = UDim2.new(0, вп.X * 0.5, 0, вп.Y * 0.5)
    else
        ФОВ_Круг.Visible = false
    end

    -- Аимбот
    ВыполнитьАимбот(дт)

    -- Прицел
    if Config.Прицел_Включен then
        local вп = Camera.ViewportSize
        Прицел.Visible    = true
        Прицел.TextSize   = Config.Прицел_Размер
        Прицел.Position   = UDim2.new(
            0, вп.X * 0.5 + Config.Прицел_ОфсетX,
            0, вп.Y * 0.5 + Config.Прицел_ОфсетY
        )

        if Config.Прицел_Вращение then
            углыСердца      = (углыСердца + Config.Прицел_Скорость) % 360
            Прицел.Rotation = углыСердца
        else
            Прицел.Rotation = 0
        end
    else
        Прицел.Visible = false
    end
end)
