    pagePadding.PaddingLeft = UDim.new(0, 8)
    pagePadding.PaddingRight = UDim.new(0, 8)

            p.Visible = false

local function CreateSlider(parent, text, min, max, default, callback)
    local frame = Instance.new("Frame", parent)
    frame.Size = UDim2.new(1, 0, 0, 46)
    frame.BackgroundColor3 = Color3.fromRGB(20, 20, 28)
    frame.ZIndex = 13

    local fCorner = Instance.new("UICorner", frame)
    fCorner.CornerRadius = UDim.new(0, 6)

    local label = Instance.new("TextLabel", frame)
    label.Size = UDim2.new(1, -60, 0, 20)
    label.Position = UDim2.new(0, 10, 0, 4)
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextColor3 = Color3.fromRGB(220, 220, 230)
    label.TextSize = 12
    label.Font = Enum.Font.Gotham
    label.TextXAlignment = Enum.TextXAlignment.Left

    local valueLabel = Instance.new("TextLabel", frame)
    valueLabel.Size = UDim2.new(0, 50, 0, 20)
    valueLabel.Position = UDim2.new(1, -55, 0, 4)
    valueLabel.BackgroundTransparency = 1
    valueLabel.Text = tostring(default)
    valueLabel.TextColor3 = Color3.fromRGB(255, 30, 60)
    valueLabel.TextSize = 12
    valueLabel.Font = Enum.Font.GothamBold

    local sliderBg = Instance.new("Frame", frame)
    sliderBg.Size = UDim2.new(1, -20, 0, 6)
    sliderBg.Position = UDim2.new(0, 10, 0, 28)
    sliderBg.BackgroundColor3 = Color3.fromRGB(40, 40, 50)

    local sBgCorner = Instance.new("UICorner", sliderBg)
    sBgCorner.CornerRadius = UDim.new(1, 0)

    local sliderFill = Instance.new("Frame", sliderBg)
    sliderFill.Size = UDim2.new(math.clamp((default - min) / (max - min), 0, 1), 0, 1, 0)
    sliderFill.BackgroundColor3 = Color3.fromRGB(255, 30, 60)

    local sFillCorner = Instance.new("UICorner", sliderFill)
    sFillCorner.CornerRadius = UDim.new(1, 0)

    local sliderBtn = Instance.new("TextButton", sliderBg)
    sliderBtn.Size = UDim2.new(1, 0, 1, 0)
    sliderBtn.BackgroundTransparency = 1
    sliderBtn.Text = ""

    local dragging = false
    sliderBtn.MouseButton1Down:Connect(function() dragging = true end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local relX = math.clamp((input.Position.X - sliderBg.AbsolutePosition.X) / sliderBg.AbsoluteSize.X, 0, 1)
            local value = math.floor(min + (max - min) * relX)
            sliderFill.Size = UDim2.new(relX, 0, 1, 0)
            valueLabel.Text = tostring(value)
            callback(value)
        end
    end)
end

local function CreateDropdown(parent, text, options, default, callback)
    local frame = Instance.new("Frame", parent)
    frame.Size = UDim2.new(1, 0, 0, 34)
    frame.BackgroundColor3 = Color3.fromRGB(20, 20, 28)
    frame.ClipsDescendants = true
    frame.ZIndex = 13

    local fCorner = Instance.new("UICorner", frame)
    fCorner.CornerRadius = UDim.new(0, 6)

    local label = Instance.new("TextLabel", frame)
    label.Size = UDim2.new(0.5, -5, 0, 34)
    label.Position = UDim2.new(0, 10, 0, 0)
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextColor3 = Color3.fromRGB(220, 220, 230)
    label.TextSize = 12
    label.Font = Enum.Font.Gotham
    label.TextXAlignment = Enum.TextXAlignment.Left

    local dropBtn = Instance.new("TextButton", frame)
    dropBtn.Size = UDim2.new(0.45, 0, 0, 24)
    dropBtn.Position = UDim2.new(0.52, 0, 0, 5)
    dropBtn.BackgroundColor3 = Color3.fromRGB(35, 35, 45)
    dropBtn.Text = default or options[1]
    dropBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    dropBtn.TextSize = 11
    dropBtn.Font = Enum.Font.Gotham

    local dCorner = Instance.new("UICorner", dropBtn)
    dCorner.CornerRadius = UDim.new(0, 4)

    local opened = false
    local optionBtns = {}

    for i, opt in ipairs(options) do
        local optBtn = Instance.new("TextButton", frame)
        optBtn.Size = UDim2.new(0.45, 0, 0, 22)
        optBtn.Position = UDim2.new(0.52, 0, 0, 34 + (i - 1) * 24)
        optBtn.BackgroundColor3 = Color3.fromRGB(28, 28, 38)
        optBtn.Text = opt
        optBtn.TextColor3 = Color3.fromRGB(200, 200, 200)
        optBtn.TextSize = 11
        optBtn.Font = Enum.Font.Gotham
        optBtn.Visible = false

        optBtn.MouseButton1Click:Connect(function()
            dropBtn.Text = opt
            opened = false
            frame.Size = UDim2.new(1, 0, 0, 34)
            for _, b in ipairs(optionBtns) do b.Visible = false end
            callback(opt)
        end)
        table.insert(optionBtns, optBtn)
    end

    dropBtn.MouseButton1Click:Connect(function()
        opened = not opened
        frame.Size = opened and UDim2.new(1, 0, 0, 34 + #options * 24 + 4) or UDim2.new(1, 0, 0, 34)
        for _, b in ipairs(optionBtns) do b.Visible = opened end
    end)
end

local ESP_GUI_Objects = {}

local function IsAlive(plr)
    if not plr or not plr.Character then return false end
    local hum = plr.Character:FindFirstChildOfClass("Humanoid")
    return hum and hum.Health > 0
end

local function IsTeammate(plr)
    if not Config.ESP_TeamCheck then return false end
    return LocalPlayer.Team and plr.Team and (LocalPlayer.Team == plr.Team)
end

local function RemovePlayerESP(plr)
    if not ESP_GUI_Objects[plr] then return end
    local o = ESP_GUI_Objects[plr]
    o.Box:Destroy()
    o.Name:Destroy()
    o.Dist:Destroy()
    o.HpBg:Destroy()
    o.Tracer:Destroy()
    ESP_GUI_Objects[plr] = nil
end

local function HideESP(o)
    o.Box.Visible = false
    o.Name.Visible = false
    o.Dist.Visible = false
    o.HpBg.Visible = false
    o.Tracer.Visible = false
end

local function UpdateESP()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LocalPlayer then continue end
        if not ESP_GUI_Objects[plr] then CreatePlayerESP(plr) end

        local o = ESP_GUI_Objects[plr]
        if not Config.ESP_Enabled or not IsAlive(plr) or IsTeammate(plr) then
            HideESP(o)
            continue
        end

        local char = plr.Character
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")

        if not hrp or not hum then
            HideESP(o)
            continue
        end

        local dist = (Camera.CFrame.Position - hrp.Position).Magnitude
        if dist > Config.ESP_MaxDistance then
            HideESP(o)
            continue
        end

        local cframe, size = char:GetBoundingBox()
        local topPos = cframe.Position + Vector3.new(0, size.Y / 2 + 0.5, 0)
        local botPos = cframe.Position - Vector3.new(0, size.Y / 2 + 0.5, 0)

        local topScreen, topOn = Camera:WorldToViewportPoint(topPos)
        local botScreen, botOn = Camera:WorldToViewportPoint(botPos)

        if not topOn or not botOn or topScreen.Z < 0 then
            HideESP(o)
            continue
        end

        local height = math.abs(botScreen.Y - topScreen.Y)
        local width = height * 0.65
        local x = topScreen.X - width / 2
        local y = topScreen.Y

        -- Box
        if Config.ESP_Boxes then
            o.Box.Visible = true
            o.Box.Position = UDim2.new(0, x, 0, y)
            o.Box.Size = UDim2.new(0, width, 0, height)
        else o.Box.Visible = false end

        -- Name
        if Config.ESP_Names then
            o.Name.Visible = true
            o.Name.Text = plr.Name
            o.Name.Position = UDim2.new(0, topScreen.X - 50, 0, y - 16)
            o.Name.Size = UDim2.new(0, 100, 0, 14)
        else o.Name.Visible = false end

        -- Distance
        if Config.ESP_Distance then
            o.Dist.Visible = true
            o.Dist.Text = math.floor(dist) .. "m"
            o.Dist.Position = UDim2.new(0, topScreen.X - 50, 0, botScreen.Y + 2)
            o.Dist.Size = UDim2.new(0, 100, 0, 12)
        else o.Dist.Visible = false end

        -- Health
        if Config.ESP_Health then
            local hpPercent = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
            o.HpBg.Visible = true
            o.HpBg.Position = UDim2.new(0, x - 6, 0, y)
            o.HpBg.Size = UDim2.new(0, 3, 0, height)

            o.HpBar.Size = UDim2.new(1, 0, hpPercent, 0)
            o.HpBar.Position = UDim2.new(0, 0, 1 - hpPercent, 0)
            o.HpBar.BackgroundColor3 = Color3.fromRGB(255 * (1 - hpPercent), 255 * hpPercent, 0)
        else o.HpBg.Visible = false end

        -- Tracer
        if Config.ESP_Tracers then
            local startX, startY = Camera.ViewportSize.X / 2, Camera.ViewportSize.Y
            local stopX, stopY = topScreen.X, botScreen.Y
            local distanceLine = math.sqrt((stopX - startX)^2 + (stopY - startY)^2)
            local angle = math.atan2(stopY - startY, stopX - startX)

            o.Tracer.Visible = true
            o.Tracer.Size = UDim2.new(0, distanceLine, 0, 1.5)
            o.Tracer.Position = UDim2.new(0, (startX + stopX) / 2, 0, (startY + stopY) / 2)
            o.Tracer.Rotation = math.deg(angle)
        else o.Tracer.Visible = false end
    end
end

local ChamsObjects = {}

local function UpdateChams()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LocalPlayer then continue end

        if Config.ESP_Enabled and Config.ESP_Chams and IsAlive(plr) and not IsTeammate(plr) then
            local char = plr.Character
            if char then
                local hl = ChamsObjects[plr] or Instance.new("Highlight")
                hl.Name = "FleexiChams"
                hl.FillColor = Config.ESP_ChamsColor
                hl.FillTransparency = Config.ESP_ChamsTransparency
                hl.OutlineColor = Color3.fromRGB(255, 30, 60)
                hl.OutlineTransparency = 0.2
                hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                hl.Parent = char
                ChamsObjects[plr] = hl
            end
        else
            if ChamsObjects[plr] then
                ChamsObjects[plr]:Destroy()
                ChamsObjects[plr] = nil
            end
        end
    end
end

-- ============================================================
-- AIMBOT SYSTEM (ZERO DELAY & NO JITTER)
-- ============================================================
local aimbotActive = false

local function GetClosestTarget()
    local closest, minDist = nil, Config.Aimbot_FOV
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LocalPlayer or not IsAlive(plr) or (Config.Aimbot_TeamCheck and IsTeammate(plr)) then continue end

        local char = plr.Character
        local part = char:FindFirstChild(Config.Aimbot_TargetPart) or char:FindFirstChild("HumanoidRootPart")
        if not part then continue end

        local screenPos, onScreen = Camera:WorldToViewportPoint(part.Position)
        if not onScreen or screenPos.Z < 0 then continue end

        local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
        if dist < minDist then
            minDist = dist
            closest = part
        end
    end
    return closest
end

local function ProcessAimbot()
    if Config.Aimbot_Enabled and Config.Aimbot_ShowFOV then
        FOVCircleGui.Visible = true
        FOVCircleGui.Size = UDim2.new(0, Config.Aimbot_FOV * 2, 0, Config.Aimbot_FOV * 2)
        FOVCircleGui.Position = UDim2.new(0, Camera.ViewportSize.X / 2, 0, Camera.ViewportSize.Y / 2)
    else
        FOVCircleGui.Visible = false
    end

    if not Config.Aimbot_Enabled or not aimbotActive then return end

    local targetPart = GetClosestTarget()
    if not targetPart then return end

    local targetPos = targetPart.Position
    if Config.Aimbot_Prediction > 0 and targetPart.Parent:FindFirstChild("HumanoidRootPart") then
        local vel = targetPart.Parent.HumanoidRootPart.AssemblyLinearVelocity
        targetPos = targetPos + vel * Config.Aimbot_Prediction
    end

    local targetCF = CFrame.new(Camera.CFrame.Position, targetPos)
    
    if Config.Aimbot_Smoothness <= 1 then
        Camera.CFrame = targetCF
    else
        Camera.CFrame = Camera.CFrame:Lerp(targetCF, 1 / Config.Aimbot_Smoothness)
    end
end

UserInputService.InputBegan:Connect(function(input, gpe)
    if gpe then return end
    if input.UserInputType == Config.Aimbot_Key or input.UserInputType == Enum.UserInputType.Touch then
        aimbotActive = true
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Config.Aimbot_Key or input.UserInputType == Enum.UserInputType.Touch then
        aimbotActive = false
    end
end)

-- ============================================================
-- CROSSHAIR SYSTEM
-- ============================================================
local crosshairPixelPool = {}
local crosshairAngle = 0

local function GetHeartGrid(density)
    local points = {}
    for i = -density, density do
        for j = -density, density do
            local x, y = i / density, j / density
            if (x*x + y*y - 1)^3 - x*x*y*y*y <= 0 then
                table.insert(points, {x = i, y = -j})
            end
        end
    end
    return points
end

local function UpdateCrosshair()
    if not Config.Crosshair_Enabled then
        CrosshairContainer.Visible = false
        return
    end
    CrosshairContainer.Visible = true

    local centerX, centerY = Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2
    local points = GetHeartGrid(Config.Crosshair_PixelDensity)
    local rad = math.rad(Config.Crosshair_Rotate and crosshairAngle or 0)
    local cosA, sinA = math.cos(rad), math.sin(rad)
    local scale = Config.Crosshair_Size / Config.Crosshair_PixelDensity

    for i, p in ipairs(points) do
        local rx = (p.x * scale) * cosA - (p.y * scale) * sinA
        local ry = (p.x * scale) * sinA + (p.y * scale) * cosA

        local px = crosshairPixelPool[i]
        if not px then
            px = Instance.new("Frame", CrosshairContainer)
            px.BorderSizePixel = 0
            crosshairPixelPool[i] = px
        end

        px.Size = UDim2.new(0, Config.Crosshair_Thickness, 0, Config.Crosshair_Thickness)
        px.Position = UDim2.new(0, centerX + rx, 0, centerY + ry)
        px.BackgroundColor3 = Config.Crosshair_Color
        px.Visible = true
    end

    for i = #points + 1, #crosshairPixelPool do
        crosshairPixelPool[i].Visible = false
    end
end
